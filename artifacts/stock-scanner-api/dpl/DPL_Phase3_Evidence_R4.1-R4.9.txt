R4.9.3 — C27 end-to-end evidence_ref validation (pseudocode)
================================================================================

  # C27: no VERIFIED row fails replay unless registered as unreplayable with verifiable evidence
  def check_c27(conn, archived_log_dir):
      cur.execute("""SELECT decision_id FROM oe_decision_audit
                     WHERE verification_status='VERIFIED' AND is_test_record=FALSE""")
      for (did,) in cur.fetchall():
          try:
              replay_decision(did, db_url=DB_URL)
          except (ReplayCodeDriftError, ReplayInputsMissingError) as exc:
              drift_class = type(exc).__name__

              # 1. Must be registered
              cur.execute("SELECT evidence_ref FROM oe_unreplayable_rows WHERE decision_id=%s", (did,))
              row = cur.fetchone()
              if row is None:
                  FAIL(f"C27: VERIFIED row {did[:16]} fails replay ({drift_class}) but not registered")
                  continue
              evidence_ref = row[0]

              # 2. Parse evidence_ref — format: "SEQ=N sha256=<64hex>"
              m = re.fullmatch(r"SEQ=(\d+) sha256=([0-9a-f]{64})", evidence_ref)
              if not m:
                  FAIL(f"C27: evidence_ref for {did[:16]} malformed: {evidence_ref!r}")
                  continue
              ref_seq, ref_sha = int(m.group(1)), m.group(2)

              # 3. Archived log must exist
              log_path = os.path.join(archived_log_dir, f"verified_run_{ref_seq}.log")
              if not os.path.exists(log_path):
                  FAIL(f"C27: archived log SEQ={ref_seq} missing for {did[:16]}")
                  continue

              # 4. sha256 must match
              actual_sha = sha256_of_file(log_path)
              if actual_sha != ref_sha:
                  FAIL(f"C27: sha256 mismatch for SEQ={ref_seq} log: stored={ref_sha[:16]} actual={actual_sha[:16]}")
                  continue

              # 5. Log must contain decision_id AND drift exception class
              log_text = open(log_path).read()
              if did not in log_text:
                  FAIL(f"C27: decision_id {did[:16]} not found in referenced log SEQ={ref_seq}")
                  continue
              if drift_class not in log_text:
                  FAIL(f"C27: exception class {drift_class} not found in referenced log SEQ={ref_seq}")
                  continue

              # All checks pass — this row is legitimately exempt
              PASS(f"C27: {did[:16]} registered + evidence verified SEQ={ref_seq}")

  Rule: a well-formed but unresolvable evidence_ref (file missing, sha256 mismatch, or
  decision_id absent from the log) causes C27 to FAIL, not skip.

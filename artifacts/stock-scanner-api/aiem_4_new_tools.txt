def _aiem_tool_signal_layer_redundancy(days_back=60, threshold=0.70):
    """
    Check whether the AIEM's signal layers are providing independent information
    or are largely echoing each other. Uses Pearson correlation across all boolean
    and numeric signals in the settled-pick log.

    A high correlation between two layers (e.g. confirmed_2d and high_conviction)
    means the conviction score is effectively double-counting that factor.
    Redundant pairs are flagged; the 'effective_independent_signals' count shows
    how many genuinely distinct signal sources exist.
    """
    try:
        import sys as _sys
        import os as _os
        _sys.path.insert(0, _os.path.dirname(__file__))
        import pandas as _pd
        from signal_correlation import run_full_correlation_report

        days_back = min(int(days_back), 120)
        threshold = float(threshold)

        with _psycopg2.connect(_DB_URL) as _c, _c.cursor() as _cu:
            _cu.execute("""
                SELECT
                    CASE WHEN confirmed_2d THEN 1 ELSE 0 END        AS confirmed_2d,
                    CASE WHEN conviction = 'HIGH' THEN 1 ELSE 0 END AS is_high_conviction,
                    COALESCE(day_ret, 0)::float                     AS gap_pct,
                    COALESCE(vol_oi, 0)::float                      AS vol_oi,
                    COALESCE(stock_price, 0)::float                 AS stock_price,
                    COALESCE(prem, 0)::float / 1000.0               AS prem_k,
                    COALESCE(otm_pct, 0)::float                     AS otm_pct,
                    COALESCE(days_out, 0)::float                    AS days_out,
                    CASE WHEN t3_win THEN 1 ELSE 0 END              AS t3_win
                FROM ai_short_calls_log
                WHERE trade_date >= CURRENT_DATE - %s
                  AND stock_price IS NOT NULL
                  AND t3_win IS NOT NULL
            """, (days_back,))
            rows = _cu.fetchall()
            cols = [d[0] for d in _cu.description]

        if not rows:
            return {"status": "no_data", "days_back": days_back}

        df = _pd.DataFrame(rows, columns=cols)
        signal_cols = ["confirmed_2d", "is_high_conviction",
                       "gap_pct", "vol_oi", "stock_price",
                       "prem_k", "otm_pct", "days_out", "t3_win"]
        report = run_full_correlation_report(df, signal_cols, threshold=threshold)

        corr_dict = report["correlation_matrix"].round(3).to_dict()
        redundant = report["redundant_pairs"].to_dict(orient="records")
        cluster   = report["cluster_summary"]

        return {
            "days_back": days_back,
            "n_picks_analyzed": len(df),
            "threshold_used": threshold,
            "correlation_matrix": corr_dict,
            "redundant_pairs": redundant,
            "cluster_summary": {
                "n_raw_signals": cluster["n_raw_signals"],
                "n_effective_independent": cluster["n_effective_clusters"],
                "clusters": cluster["clusters"],
            },
            "interpretation": (
                f"{cluster['n_effective_clusters']} truly independent signal clusters "
                f"from {cluster['n_raw_signals']} raw signals. "
                + ("No redundancy found above threshold." if not redundant else
                   "Redundant pairs: " + ", ".join(
                       r["signal_a"] + "\u2194" + r["signal_b"] +
                       " (" + format(r["correlation"], "+.2f") + ")"
                       for r in redundant[:5]
                   ))
            ),
            "next_step": (
                "If n_effective_independent < n_raw_signals: weight redundant signals lower "
                "in test_scoring_hypothesis (they add less marginal information). "
                "If a cluster has 3+ signals all measuring the same thing, that cluster is "
                "only worth ~1 vote in the conviction score, not 3."
            )
        }
    except Exception as e:
        return {"status": "error", "error": str(e)}

def _aiem_tool_signal_magnitude_analysis(days_back=60, n_bins=6):
    """
    Check whether win rate scales continuously with signal magnitude or plateaus.
    Tests gap_pct, vol_oi, and stock_price across equal-frequency bins.

    A monotonically increasing relationship means 'more = better' — supports
    raising the minimum threshold. A plateau or peak-in-middle means the current
    binary cutoff may already be near-optimal; extreme values may not add edge.
    """
    try:
        import sys as _sys
        import os as _os
        _sys.path.insert(0, _os.path.dirname(__file__))
        import pandas as _pd
        from signal_magnitude_analysis import magnitude_report

        days_back = min(int(days_back), 120)
        n_bins    = max(4, min(int(n_bins), 10))

        with _psycopg2.connect(_DB_URL) as _c, _c.cursor() as _cu:
            _cu.execute("""
                SELECT
                    COALESCE(day_ret, 0)::float       AS gap_pct,
                    COALESCE(vol_oi, 0)::float        AS vol_oi,
                    COALESCE(stock_price, 0)::float   AS stock_price,
                    CASE WHEN t3_win THEN 1 ELSE 0 END AS outcome
                FROM ai_short_calls_log
                WHERE trade_date >= CURRENT_DATE - %s
                  AND t3_win IS NOT NULL
                  AND stock_price > 0
            """, (days_back,))
            rows = _cu.fetchall()
            cols = [d[0] for d in _cu.description]

        if not rows:
            return {"status": "no_data", "days_back": days_back}

        df = _pd.DataFrame(rows, columns=cols).dropna()
        report = magnitude_report(df, ["gap_pct", "vol_oi", "stock_price"],
                                  outcome_col="outcome", n_bins=n_bins)

        output = {"days_back": days_back, "n_settled": len(df), "signals": {}}
        for col, result in report.items():
            if result["status"] != "ok":
                output["signals"][col] = {"status": result["status"]}
                continue
            tbl = result["table"]
            output["signals"][col] = {
                "shape": result["shape"],
                "bins": [
                    {
                        "range": f"{row['signal_min']:.2f}–{row['signal_max']:.2f}",
                        "mean_value": round(float(row["signal_mean"]), 2),
                        "n": int(row["n"]),
                        "win_rate_pct": round(float(row["win_rate"]) * 100, 1),
                    }
                    for _, row in tbl.iterrows()
                ],
            }
        output["next_step"] = (
            "For signals with 'monotonically_increasing' shape: raising the minimum "
            "threshold (e.g. vol_oi >= 5 instead of >= 2) should increase win rate. "
            "For 'peaks_in_middle': current threshold may already be near-optimal — "
            "do NOT raise it further without retesting. Run test_new_signal to validate."
        )
        return output
    except Exception as e:
        return {"status": "error", "error": str(e)}

def _aiem_tool_holding_period_optimize(days_back=60, segment_by="conviction"):
    """
    Compare win rate and average return at T+1, T+3, and T+5 exit points,
    broken down by conviction level or signal type. Determines whether holding
    longer is systematically worth it — or whether early exits are better.

    Note: data has T+1, T+3, T+5 checkpoints (not every day). The tool
    identifies the best of these three horizons per segment.
    """
    try:
        import sys as _sys
        import os as _os
        _sys.path.insert(0, _os.path.dirname(__file__))
        import pandas as _pd
        import numpy as _np
        from holding_period_optimizer import aggregate_horizon_performance, find_optimal_horizon

        days_back  = min(int(days_back), 120)
        seg_col    = segment_by if segment_by in ("conviction", "confirmed_2d") else "conviction"

        with _psycopg2.connect(_DB_URL) as _c, _c.cursor() as _cu:
            _cu.execute("""
                SELECT
                    id,
                    conviction,
                    confirmed_2d,
                    t1_pct::float  AS t1_pct,
                    t3_pct::float  AS t3_pct,
                    t5_pct::float  AS t5_pct,
                    CASE WHEN t3_win THEN 1 ELSE 0 END AS t3_win
                FROM ai_short_calls_log
                WHERE trade_date >= CURRENT_DATE - %s
                  AND (t1_pct IS NOT NULL OR t3_pct IS NOT NULL OR t5_pct IS NOT NULL)
            """, (days_back,))
            rows = _cu.fetchall()
            cols = [d[0] for d in _cu.description]

        if not rows:
            return {"status": "no_data", "days_back": days_back}

        df = _pd.DataFrame(rows, columns=cols)

        # Reshape to long format via melt — O(n) not O(n*3) row-by-row
        _df_melt = _pd.melt(
            df,
            id_vars=["id", seg_col],
            value_vars=["t1_pct", "t3_pct", "t5_pct"],
            var_name="_hcol",
            value_name="return_raw",
        ).dropna(subset=["return_raw"])
        _hmap = {"t1_pct": 1, "t3_pct": 3, "t5_pct": 5}
        _df_melt["horizon"]    = _df_melt["_hcol"].map(_hmap)
        _df_melt["return_pct"] = _df_melt["return_raw"] / 100.0
        _df_melt["pick_id"]    = _df_melt["id"]
        long_df = _df_melt[["pick_id", "horizon", "return_pct", seg_col]].copy()

        # Overall best horizon
        overall_perf = aggregate_horizon_performance(long_df)
        best_overall = find_optimal_horizon(overall_perf, optimize_for="avg_return_pct")

        # By segment
        seg_results = {}
        for seg_val in df[seg_col].dropna().unique():
            seg_long = long_df[long_df[seg_col] == seg_val]
            if len(seg_long) < 10:
                continue
            perf = aggregate_horizon_performance(seg_long)
            best = find_optimal_horizon(perf, optimize_for="avg_return_pct")
            seg_results[str(seg_val)] = {
                "best_horizon_days": best["best_horizon"],
                "avg_return_at_best_pct": round(best["metric_value"] * 100, 2),
                "n_picks": best["n_samples_at_best_horizon"],
                "horizons": [
                    {
                        "days": int(r["horizon"]),
                        "n": int(r["n"]),
                        "win_rate_pct": round(float(r["win_rate"]) * 100, 1),
                        "avg_return_pct": round(float(r["avg_return_pct"]) * 100, 2),
                    }
                    for _, r in perf.iterrows()
                ],
            }

        return {
            "days_back": days_back,
            "n_picks_analyzed": len(df),
            "segment_by": seg_col,
            "best_overall_horizon_days": best_overall["best_horizon"],
            "by_segment": seg_results,
            "next_step": (
                "If HIGH conviction picks peak at T+1 but MEDIUM peaks at T+3: "
                "grade HIGH picks at T+1 and MEDIUM at T+3 in the retrain model. "
                "Run run_statistical_significance to confirm each difference is real "
                "before hard-coding the split."
            )
        }
    except Exception as e:
        return {"status": "error", "error": str(e)}

def _aiem_tool_kelly_position_size(days_back=60, fractional_multiplier=0.25,
                                   segment_by=None, horizon=3):
    """
    Compute quarter-Kelly optimal position sizing from settled pick outcomes.
    Lower multiplier when sample size is small (< 200 picks).

    Returns the recommended fraction of capital per trade — NOT a dollar amount.
    Segmented by conviction level if segment_by='conviction'.

    horizon — which exit to use for the return distribution (1, 3, or 5 days).
    Match this to the optimal exit found by holding_period_optimize: if HIGH
    conviction picks optimally exit at T+1, pass horizon=1 so Kelly is computed
    on the correct return distribution for that segment, not on T+3 returns.

    IMPORTANT: Only use this AFTER a signal has cleared statistical significance
    in run_statistical_significance. Sizing around an unvalidated edge amplifies
    losses, not gains.
    """
    try:
        import sys as _sys
        import os as _os
        _sys.path.insert(0, _os.path.dirname(__file__))
        import pandas as _pd
        from position_sizing import kelly_from_settled_picks

        days_back   = min(int(days_back), 180)
        frac_mult   = max(0.1, min(float(fractional_multiplier), 0.5))
        _valid_segs = ("conviction", "confirmed_2d")
        if segment_by is not None and segment_by not in _valid_segs:
            print(f"[kelly_position_size] unrecognized segment_by={segment_by!r}; "
                  f"valid: {_valid_segs} — segmentation skipped")
        seg_col = segment_by if segment_by in _valid_segs else None

        # Map horizon → DB columns
        _hor = int(horizon) if int(horizon) in (1, 3, 5) else 3
        if _hor != int(horizon):
            print(f"[kelly_position_size] unrecognized horizon={horizon!r}; valid: 1/3/5 — using T+3")
        _win_col = f"t{_hor}_win"
        _pct_col = f"t{_hor}_pct"

        with _psycopg2.connect(_DB_URL) as _c, _c.cursor() as _cu:
            _cu.execute(f"""
                SELECT
                    conviction,
                    confirmed_2d,
                    {_win_col}::int                        AS outcome,
                    COALESCE({_pct_col}, 0)::float / 100.0 AS return_pct
                FROM ai_short_calls_log
                WHERE trade_date >= CURRENT_DATE - %s
                  AND {_win_col} IS NOT NULL
                  AND {_pct_col} IS NOT NULL
            """, (days_back,))
            rows = _cu.fetchall()
            cols = [d[0] for d in _cu.description]

        if not rows:
            return {"status": "no_data", "days_back": days_back}

        df = _pd.DataFrame(rows, columns=cols)

        def _fmt(kr):
            return {
                "win_rate_pct":          round(kr.win_rate * 100, 1),
                "avg_win_pct":           round(kr.avg_win_pct * 100, 2),
                "avg_loss_pct":          round(kr.avg_loss_pct * 100, 2),
                "payoff_ratio":          round(float(kr.payoff_ratio), 2),
                "full_kelly_fraction":   round(kr.full_kelly_fraction, 4),
                "recommended_fraction":  round(kr.recommended_fraction, 4),
                "recommended_pct_capital": round(kr.recommended_fraction * 100, 2),
                "fractional_multiplier": round(kr.fractional_kelly_multiplier, 4),
                "warning":               kr.warning,
            }

        result = {
            "days_back": days_back,
            "n_settled_picks": len(df),
            "fractional_multiplier_requested": frac_mult,
        }

        result["overall"] = _fmt(kelly_from_settled_picks(df, fractional_multiplier=frac_mult))

        if seg_col:
            result["by_segment"] = {}
            for val in df[seg_col].dropna().unique():
                seg_df = df[df[seg_col] == val]
                if len(seg_df) < 20:
                    continue
                result["by_segment"][str(val)] = {
                    "n": len(seg_df),
                    **_fmt(kelly_from_settled_picks(seg_df, fractional_multiplier=frac_mult))
                }

        result["next_step"] = (
            "recommended_pct_capital is the quarter-Kelly bet size per trade. "
            "If win rate < 50% or payoff_ratio < 1.0, full_kelly_fraction will be zero or "
            "negative — no edge justifies a position. "
            "Segment by conviction='HIGH' vs 'MEDIUM' to size HIGH picks larger if their "
            "edge is proven significant."
        )
        return result
    except Exception as e:
        return {"status": "error", "error": str(e)}

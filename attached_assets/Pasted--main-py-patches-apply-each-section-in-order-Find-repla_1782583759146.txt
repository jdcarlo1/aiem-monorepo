def _research(q=question, to=from_num):
    session_name = f"sms_{q[:20].replace(' ','_')}_{uuid.uuid4().hex[:6]}"
    prompt = (
        f"The owner just texted this question: '{q}'\n\n"
        f"Research this using your tools. If they mention tickers, use "
        f"mkt_retrospective_backtest and mkt_find_behavioral_matches. "
        f"If they ask why stocks moved, use mkt_analyze_top_movers + mkt_retrospective_backtest. "
        f"If they ask about a signal or pattern, use mkt_test_signal to validate. "
        f"After researching, write a SHORT reply (under 300 chars) that directly answers "
        f"their question with the most important finding. Start with the direct answer."
    )

    # Reuse the SAME lock the email "ask AIEM" path uses, so SMS, email,
    # web chat, and scheduled cron sessions can never run concurrently
    # against each other and step on shared resources / cost budget.
    if not app._aiem_qa_lock.acquire(blocking=False):
        _send_sms("⏳ Already researching another question — I'll get to this one next. Try again in a few minutes.", to=to)
        return

    try:
        _run_aiem_focused_session(
            session_name=session_name,
            focus_prompt=prompt,
            max_iterations=8
        )
        # Pull THIS session's insight specifically, not just "the latest one" —
        # prevents reading back a different concurrent session's answer.
        try:
            import psycopg2 as _pg
            with _pg.connect(os.environ["DATABASE_URL"]) as _c, _c.cursor() as _cur:
                _cur.execute(
                    """SELECT insight_text FROM aiem_research_insights
                       WHERE session_name=%s ORDER BY created_at DESC LIMIT 1""",
                    (session_name,),
                )
                row = _cur.fetchone()
                if not row:
                    # fall back to latest overall only if nothing tagged to this session
                    _cur.execute("SELECT insight_text FROM aiem_research_insights ORDER BY created_at DESC LIMIT 1")
                    row = _cur.fetchone()
                summary = (row[0][:280] if row and row[0] else
                           "Research complete — check your email for the full report.")
        except Exception:
            summary = "Research complete — check your email for the full report."
        _send_sms(f"📊 {summary}", to=to)
    except Exception as _e:
        print(f"[sms/incoming] research error: {_e}")
        _send_sms("⚠️ Hit an error researching that. Try again.", to=to)
    finally:
        app._aiem_qa_lock.release()

import json

def _aiem_classify_question_complexity(question: str) -> int:
    """Dynamic iteration budget instead of a flat number — complex
    multi-step questions get more room before the model has to answer."""
    q = question.lower()
    deep_signals = ["backtest", "validate", "significan", "out-of-sample",
                     "oos", "is it real", "does it work", "prove", "p-value"]
    compare_signals = ["compare", "vs", "versus", "better than", "which of"]
    simple_signals = ["what is", "define", "explain", "how does"]
    if any(s in q for s in deep_signals):
        return 10
    if any(s in q for s in compare_signals):
        return 7
    if any(s in q for s in simple_signals) and len(q) < 80:
        return 4
    return 6


def _get_thread_db_conn():
    """New connection per thread — never reuse a connection across threads.
    Swap this for your existing pool accessor if you have one."""
    import psycopg2
    return psycopg2.connect(os.environ["DATABASE_URL"])


@app.route("/stock-api/aiem/chat", methods=["POST"])
def aiem_chat_start():
    data = request.get_json(silent=True) or {}
    question = (data.get("question") or "").strip()[:800]
    if not question:
        return jsonify({"error": "question is required"}), 400

    # Single-flight check against the SAME lock used by SMS/email, so the
    # web chat tab can't run a session concurrently with those either.
    if not app._aiem_qa_lock.acquire(blocking=False):
        return jsonify({
            "error": "Another research session is already running (SMS, email, "
                      "or chat). Please wait for it to finish before starting a new one.",
        }), 429
    app._aiem_qa_lock.release()  # release immediately — the worker thread re-acquires
                                   # it itself so the HTTP request doesn't hold the lock

    job_id = str(_uuid_mod.uuid4())
    conn = _get_thread_db_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(
                """INSERT INTO quant_agent_sessions (job_id, question, status)
                   VALUES (%s, %s, 'pending')""",
                (job_id, question),
            )
        conn.commit()
    finally:
        conn.close()

    max_iter = _aiem_classify_question_complexity(question)
    prompt = (
        f"The user asks: '{question}'\n\n"
        f"Research this thoroughly using your tools. "
        f"If they mention specific tickers, use mkt_retrospective_backtest. "
        f"If they ask about a pattern or signal, use mkt_test_signal to validate with real data. "
        f"End with a clear, direct answer. Be concise but complete — 3-5 paragraphs max."
    )

    def _worker():
        if not app._aiem_qa_lock.acquire(blocking=False):
            wconn = _get_thread_db_conn()
            try:
                with wconn.cursor() as cur:
                    cur.execute(
                        "UPDATE quant_agent_sessions SET status='error', error=%s WHERE job_id=%s",
                        ("Another session started first — please retry.", job_id),
                    )
                wconn.commit()
            finally:
                wconn.close()
            return

        worker_conn = _get_thread_db_conn()
        try:
            with worker_conn.cursor() as cur:
                cur.execute(
                    "UPDATE quant_agent_sessions SET status='running', updated_at=NOW() WHERE job_id=%s",
                    (job_id,),
                )
            worker_conn.commit()

            def _on_step(step):
                try:
                    with worker_conn.cursor() as cur:
                        cur.execute(
                            "UPDATE quant_agent_sessions SET current_tool=%s, updated_at=NOW() WHERE job_id=%s",
                            (step.get("tool"), job_id),
                        )
                    worker_conn.commit()
                except Exception:
                    pass

            answer, trace, err = _run_aiem_focused_session_v2(
                session_name=f"quant_chat_{job_id[:8]}",
                focus_prompt=prompt,
                max_iterations=max_iter,
                on_step=_on_step,
            )

            with worker_conn.cursor() as cur:
                if err:
                    cur.execute(
                        """UPDATE quant_agent_sessions
                           SET status='error', error=%s, tool_trace=%s, updated_at=NOW()
                           WHERE job_id=%s""",
                        (err, json.dumps(trace), job_id),
                    )
                else:
                    cur.execute(
                        """UPDATE quant_agent_sessions
                           SET status='done', answer=%s, tool_trace=%s, updated_at=NOW()
                           WHERE job_id=%s""",
                        (answer, json.dumps(trace), job_id),
                    )
            worker_conn.commit()

        except Exception as e:
            try:
                with worker_conn.cursor() as cur:
                    cur.execute(
                        "UPDATE quant_agent_sessions SET status='error', error=%s, updated_at=NOW() WHERE job_id=%s",
                        (f"worker crashed: {type(e).__name__}: {e}", job_id),
                    )
                worker_conn.commit()
            except Exception:
                pass
        finally:
            worker_conn.close()
            app._aiem_qa_lock.release()

    threading.Thread(target=_worker, daemon=True).start()
    return jsonify({"job_id": job_id, "status": "pending"})


@app.route("/stock-api/aiem/chat/<job_id>", methods=["GET"])
def aiem_chat_poll(job_id):
    conn = _get_thread_db_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(
                """SELECT status, answer, error, current_tool, tool_trace, created_at
                   FROM quant_agent_sessions WHERE job_id=%s""",
                (job_id,),
            )
            row = cur.fetchone()
    finally:
        conn.close()

    if row is None:
        return jsonify({"error": "job not found"}), 404

    status, answer, error, current_tool, tool_trace, created_at = row
    return jsonify({
        "job_id": job_id, "status": status, "answer": answer, "error": error,
        "current_tool": current_tool, "tool_trace": tool_trace,
        "created_at": created_at.isoformat() if created_at else None,
    })


@app.route("/stock-api/aiem/chat/history", methods=["GET"])
def aiem_chat_history():
    conn = _get_thread_db_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(
                """SELECT job_id, question, status, answer, tool_trace, created_at
                   FROM quant_agent_sessions ORDER BY created_at DESC LIMIT 20"""
            )
            rows = cur.fetchall()
    finally:
        conn.close()

    return jsonify([
        {"job_id": r[0], "question": r[1], "status": r[2], "answer": r[3],
         "tool_trace": r[4], "created_at": r[5].isoformat() if r[5] else None}
        for r in rows
    ])


def reconcile_orphaned_chat_sessions():
    """Call once at app startup — recovers jobs stuck at 'running' if the
    process restarted mid-session (redeploy, crash, autoscale cycle)."""
    conn = _get_thread_db_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(
                """UPDATE quant_agent_sessions
                   SET status='error', error='orphaned: server restarted mid-session'
                   WHERE status='running' AND updated_at < NOW() - INTERVAL '10 minutes'"""
            )
        conn.commit()
    finally:
        conn.close()

import time as _time_mod


def _run_aiem_focused_session_v2(session_name: str, focus_prompt: str,
                                  max_iterations: int = 12, on_step=None):
    from openai import OpenAI as _OAIFS

    _oai = _OAIFS(
        base_url=os.environ.get("AI_INTEGRATIONS_OPENAI_BASE_URL", "https://ai-integrations.replit.com/openai"),
        api_key=os.environ.get("AI_INTEGRATIONS_OPENAI_API_KEY", ""),
    )

    _fs_schema = _AIEM_AGENT_TOOLS  # the existing shared schema, unchanged

    session_system = (
        _AIEM_AGENT_SYSTEM +
        f"\n\nSESSION FOCUS ({session_name}): {focus_prompt}\n"
        "Start immediately with the most relevant tools for this session's focus."
    )
    messages = [
        {"role": "system", "content": session_system},
        {"role": "user", "content": focus_prompt},
    ]

    trace = []
    _last_text = ""
    _context_budget = 60_000
    _context_used = 0

    for _i in range(max_iterations):
        resp = None
        for attempt in range(3):
            try:
                resp = _oai.chat.completions.create(
                    model="gpt-4o", messages=messages, tools=_fs_schema,
                    tool_choice="auto", max_tokens=2000, temperature=0.3,
                )
                break
            except Exception as e:
                if attempt == 2:
                    return _last_text, trace, f"OpenAI call failed after retries: {e}"
                _time_mod.sleep(1.5 * (attempt + 1))

        msg = resp.choices[0].message
        msg_dict = {"role": "assistant", "content": msg.content or None}
        if msg.tool_calls:
            msg_dict["tool_calls"] = [
                {"id": tc.id, "type": "function",
                 "function": {"name": tc.function.name, "arguments": tc.function.arguments}}
                for tc in msg.tool_calls
            ]
        messages.append(msg_dict)

        if msg.content:
            _last_text = msg.content
        if not msg.tool_calls:
            break

        for tc in msg.tool_calls:
            fn = tc.function.name
            try:
                args = json.loads(tc.function.arguments or "{}")
            except json.JSONDecodeError as e:
                args = None
                result = {"error": f"malformed arguments from model: {e}"}

            if args is not None:
                if fn not in _fs_tool_map:
                    result = {"error": f"unknown tool: {fn}"}
                else:
                    try:
                        result = _fs_tool_map[fn](**args)
                    except Exception as e:
                        result = {"error": f"{fn} raised: {type(e).__name__}: {e}"}

            result_str = json.dumps(result, default=str)
            if len(result_str) > 6000:
                result_str = result_str[:6000] + "...[truncated]"
            _context_used += len(result_str)
            if _context_used > _context_budget:
                result_str = json.dumps({"note": "result omitted, context budget reached"})

            step = {"iteration": _i, "tool": fn, "args": args,
                    "ok": "error" not in (result if isinstance(result, dict) else {})}
            trace.append(step)
            if on_step:
                try:
                    on_step(step)
                except Exception:
                    pass

            messages.append({"role": "tool", "tool_call_id": tc.id, "content": result_str})

    return _last_text, trace, None

# SECTION 6 — NEW TOOL: aiem_tool_get_daily_candidates
#
# Answers exactly: "give me N stocks today that look ready to move, based
# on current indicators and historically similar setups."
#
# This does NOT invent new signals or new alpha. It's a read-only join of
# two things you already built and already run on a schedule:
#   - conviction_stack_watchlist (8-layer conviction score, snapshotted daily)
#   - behavioral_pattern_matches (mkt_find_behavioral_matches' underlying
#     table — current setup vs. historical pre-move fingerprints)
#
# Without this, the chat agent has to correctly chain 2-3 separate tool
# calls itself every time someone asks this very common question, and rank/
# dedupe the results coherently on its own. That's exactly the kind of
# multi-step task where tool-selection luck causes inconsistent answers.
# This tool makes the combination deterministic and always includes the
# statistical backing (sample size, similarity score, what the historical
# analog actually did) rather than letting the model improvise the framing.
#
# Add the function below anywhere your other _aiem_tool_* functions live.
# ============================================================================

def _aiem_tool_get_daily_candidates(limit: int = 10, min_conviction_pct: int = 60,
                                     min_similarity: float = 0.75) -> dict:
    """Get today's top N candidates ranked by conviction score, cross-referenced
    against historical behavioral pattern matches. This is the tool to use
    whenever someone asks for 'stocks today that look ready to move', 'picks
    for today', or similar — it combines the daily conviction stack snapshot
    with the behavioral pre-move matcher and returns ranked candidates with
    their historical analog (what similar setups did historically), not a
    prediction. Always present results as pattern-match probability with
    sample size, never as a guarantee.
    """
    import psycopg2

    try:
        with psycopg2.connect(os.environ["DATABASE_URL"]) as conn, conn.cursor() as cur:
            # Today's conviction stack snapshot, ranked
            cur.execute(
                """SELECT ticker, total_pts, conviction_pct, label, price, layers, rank
                   FROM conviction_stack_watchlist
                   WHERE snap_date = CURRENT_DATE AND conviction_pct >= %s
                   ORDER BY conviction_pct DESC, total_pts DESC
                   LIMIT %s""",
                (min_conviction_pct, limit * 3),  # over-fetch; behavioral match filters some out
            )
            conviction_rows = cur.fetchall()

            if not conviction_rows:
                return {
                    "status": "ok", "candidates": [], "n": 0,
                    "note": (
                        f"No tickers in conviction_stack_watchlist for today met the "
                        f"conviction_pct >= {min_conviction_pct} bar. Either today's "
                        f"snapshot hasn't run yet, or nothing currently clears the bar — "
                        f"lower min_conviction_pct to see weaker candidates, or report "
                        f"honestly that nothing strong qualifies today rather than "
                        f"lowering the bar just to produce a list."
                    ),
                }

            tickers = [r[0] for r in conviction_rows]

            # Behavioral matches for those same tickers, most recent scan
            cur.execute(
                """SELECT DISTINCT ON (ticker) ticker, similarity, matched_ticker,
                          matched_date, matched_move, days_before_move, verdict, scan_time
                   FROM behavioral_pattern_matches
                   WHERE ticker = ANY(%s) AND scan_time >= NOW() - INTERVAL '2 hours'
                   ORDER BY ticker, similarity DESC""",
                (tickers,),
            )
            match_rows = {r[0]: r for r in cur.fetchall()}

        candidates = []
        for ticker, total_pts, conviction_pct, label, price, layers, rank in conviction_rows:
            match = match_rows.get(ticker)
            entry = {
                "ticker": ticker,
                "conviction_pct": conviction_pct,
                "conviction_label": label,
                "price": float(price) if price is not None else None,
                "layer_breakdown": layers,
                "behavioral_match": None,
            }
            if match:
                _, similarity, matched_ticker, matched_date, matched_move, days_before, verdict, scan_time = match
                if float(similarity) >= min_similarity:
                    entry["behavioral_match"] = {
                        "similarity": float(similarity),
                        "historical_analog_ticker": matched_ticker,
                        "historical_analog_date": str(matched_date),
                        "historical_analog_move_pct": float(matched_move or 0),
                        "days_before_that_move": days_before,
                        "verdict": verdict,
                    }
            candidates.append(entry)

        # Prioritize candidates that have BOTH high conviction AND a behavioral
        # match — that's the strongest combined signal, not either alone.
        candidates.sort(
            key=lambda c: (c["behavioral_match"] is not None, c["conviction_pct"]),
            reverse=True,
        )
        candidates = candidates[:limit]

        n_with_match = sum(1 for c in candidates if c["behavioral_match"])

        return {
            "status": "ok",
            "candidates": candidates,
            "n": len(candidates),
            "n_with_historical_match": n_with_match,
            "framing_required": (
                "When presenting this to a user: state these are pattern-similarity "
                "matches against historical pre-move setups, not predictions. Always "
                "include the similarity score and what the historical analog actually "
                "did. If fewer than the requested number of candidates clear the bar, "
                "say so explicitly rather than padding the list with weaker names."
            ),
        }
    except Exception as e:
        return {"status": "error", "error": str(e)}

# SECTION 8 — NEW TOOL: get_live_snapshot — true intraday data, fetched at
# the moment the question is asked, not from a once-daily or 30-min-cadence
# cached table.
#
# IMPORTANT CAVEAT: whether this is real-time or delayed depends on your
# Polygon subscription tier. Polygon's free/basic tiers deliver 15-minute
# delayed data on this endpoint; real-time requires their paid Stocks
# Starter/Developer plan or above. Check what plan POLYGON_API_KEY is
# tied to before assuming this is truly live.
#
# Uses the same urllib + POLYGON_API_KEY pattern already used for the
# gainers scan (~line 28783) — no new auth/dependency, just a different
# endpoint, called on-demand instead of on a schedule.
# ============================================================================

def _aiem_tool_get_live_snapshot(tickers) -> dict:
    """Get live (or near-live, depending on your Polygon plan) intraday data
    for specific tickers RIGHT NOW — current price, today's volume so far,
    the most recent minute bar, and change % from previous close. Use this
    whenever a question asks about 'right now', 'currently', 'this minute',
    or otherwise needs fresher data than the daily conviction snapshot or
    the 30-min behavioral scan can provide. Pass a list of ticker symbols
    (max 50 per call to keep the response size sane).
    """
    import os as _os_ls, urllib.request as _ureq_ls, urllib.parse as _uparse_ls, json as _json_ls

    if isinstance(tickers, str):
        tickers = [tickers]
    tickers = [t.upper().strip() for t in tickers if t][:50]
    if not tickers:
        return {"status": "error", "error": "no tickers provided"}

    key = _os_ls.environ.get("POLYGON_API_KEY", "")
    if not key:
        return {"status": "error", "error": "POLYGON_API_KEY not configured"}

    ticker_param = _uparse_ls.quote(",".join(tickers))
    url = (
        f"https://api.polygon.io/v2/snapshot/locale/us/markets/stocks/tickers"
        f"?tickers={ticker_param}&apiKey={key}"
    )

    try:
        with _ureq_ls.urlopen(url, timeout=8) as resp:
            data = _json_ls.load(resp)
    except Exception as e:
        return {"status": "error", "error": f"Polygon snapshot fetch failed: {e}"}

    if data.get("status") not in ("OK", "DELAYED"):
        return {"status": "error", "error": f"Polygon returned status={data.get('status')}"}

    results = []
    for t in (data.get("tickers") or []):
        day = t.get("day") or {}
        min_bar = t.get("min") or {}
        prev = t.get("prevDay") or {}
        last_trade = t.get("lastTrade") or {}
        results.append({
            "ticker": t.get("ticker"),
            "last_price": last_trade.get("p") or day.get("c"),
            "change_pct_today": t.get("todaysChangePerc"),
            "day_open": day.get("o"),
            "day_high": day.get("h"),
            "day_low": day.get("l"),
            "day_volume_so_far": day.get("v"),
            "most_recent_minute_bar": {
                "open": min_bar.get("o"), "high": min_bar.get("h"),
                "low": min_bar.get("l"), "close": min_bar.get("c"),
                "volume": min_bar.get("v"),
                "bar_timestamp": min_bar.get("t"),
            } if min_bar else None,
            "prev_close": prev.get("c"),
            "snapshot_updated_at": t.get("updated"),
        })

    return {
        "status": "ok",
        "data_status": data.get("status"),  # "OK" (real-time, your plan dependent) or "DELAYED"
        "snapshots": results,
        "n": len(results),
        "note": (
            "If data_status is DELAYED, this is NOT real-time — it reflects "
            "Polygon's delayed-data tier. Say so explicitly if asked for "
            "real-time data and this comes back delayed, rather than "
            "presenting delayed data as current."
        ),
    }

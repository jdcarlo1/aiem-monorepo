#!/bin/bash
# verify_aiem_actual_reasoning.sh
# This checks what AIEM's DEPLOYED, PRODUCTION code actually reasoned
# about real picks — not what the Replit coding chatbot said when asked
# a theory question. These are two different things. This script proves
# (or disproves) that the real running system's logic matches sound
# theory, using its own logged decisions as evidence.

echo "################################################################"
echo "# AIEM'S ACTUAL LOGGED REASONING (production system, not chat)"
echo "################################################################"

if [ -z "$DATABASE_URL" ]; then
  echo "DATABASE_URL not set."
  exit 1
fi

echo ""
echo "=== STEP 1: Pull real reasoning text from actual past decisions ==="
echo "(These are AIEM's own words, written by the deployed code when it"
echo " made a REAL decision — not a chatbot answering a question.)"
psql "$DATABASE_URL" -c "
  SELECT decision_time, signal_name, decision_type, ticker,
         LEFT(reasoning, 300) AS reasoning_excerpt
  FROM agent_decisions
  ORDER BY decision_time DESC
  LIMIT 15;"

echo ""
echo "=== STEP 2: Specifically pull any gamma/charm/dark-pool related decisions ==="
psql "$DATABASE_URL" -c "
  SELECT decision_time, ticker, signal_name,
         LEFT(reasoning, 400) AS reasoning_excerpt
  FROM agent_decisions
  WHERE reasoning ILIKE '%gamma%'
     OR reasoning ILIKE '%charm%'
     OR reasoning ILIKE '%dark pool%'
  ORDER BY decision_time DESC
  LIMIT 15;"

echo ""
echo "=== STEP 3: Check whether reasoning is genuinely distinct per decision ==="
echo "(If every row has near-identical reasoning text, that's a sign the"
echo " production system is templating output, not reasoning per-case —"
echo " same red flag as the fake log lines from earlier.)"
psql "$DATABASE_URL" -c "
  SELECT COUNT(*) AS total_decisions,
         COUNT(DISTINCT reasoning) AS distinct_reasoning_strings
  FROM agent_decisions
  WHERE reasoning IS NOT NULL AND reasoning != '';"

echo ""
echo "=== STEP 4: Locate the ACTUAL code that generates this reasoning text ==="
echo "(This shows us where in the real codebase the reasoning string gets"
echo " built — confirming whether it's computed dynamically from real"
echo " signal values, or hardcoded/templated.)"
grep -rn "reasoning\s*=" --include="*.py" . | grep -v "test\|__pycache__" | head -30

echo ""
echo "################################################################"
echo "# DONE — paste this entire output back unedited"
echo "################################################################"
echo "# What this proves: whether AIEM's REAL, DEPLOYED decision logic"
echo "# produces reasoning consistent with sound theory on actual past"
echo "# picks — which is different from, and more trustworthy than,"
echo "# a chat conversation answering a theory question."

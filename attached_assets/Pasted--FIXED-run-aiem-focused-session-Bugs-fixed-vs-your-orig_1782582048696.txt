"""
FIXED: _run_aiem_focused_session

Bugs fixed vs. your original:
1. `messages.append(msg)` appended the raw OpenAI SDK message object. This
   sometimes round-trips fine, but tool_calls on that object are SDK types,
   not plain dicts — on some openai-python versions this throws on the NEXT
   request when it gets re-serialized, or silently drops fields. Always
   convert to a plain dict before appending to `messages`.
2. No try/except around tool execution. If ANY of your 88 tools throws
   (bad ticker, DB timeout, None from a missing row, etc.) the whole
   session dies, the exception propagates up through the daemon thread,
   and Flask never gets a chance to mark the job 'error'. The job sits at
   'running' forever and the frontend polls it forever. Fixed by catching
   per-tool-call and feeding the error back to the model as a tool result
   (the model can usually recover and try a different tool).
3. No try/except around json.loads(tc.function.arguments) — gpt-4o
   occasionally emits malformed JSON for complex args. Same failure mode
   as #2.
4. No retry/backoff on the OpenAI call itself (rate limits, transient
   network errors). Added a small retry loop.
5. No visibility into which tools were called — added a `trace` list that
   the caller can persist to the DB (answers your improvement-area #6).
6. No cap on total tool-result bytes fed back into context across
   iterations — 12 iterations * 6000 chars per tool result can still add
   up. Added a running budget so a long session doesn't blow context.
"""
import os
import json
import time


def _run_aiem_focused_session(session_name: str, focus_prompt: str,
                               max_iterations: int = 12,
                               on_step=None):
    """
    Parameterized research session. The agent uses all its tools but starts
    with a specific focus question. Returns (final_text, trace, error).

    on_step: optional callback(step_dict) called after each tool call,
             so the Flask worker can persist live progress to the DB for
             the frontend to show ("Calling mkt_test_signal...") instead of
             just a generic spinner.
    """
    from openai import OpenAI as _OAIFS

    _oai = _OAIFS(
        base_url=os.environ.get("AI_INTEGRATIONS_OPENAI_BASE_URL"),
        api_key=os.environ.get("AI_INTEGRATIONS_OPENAI_API_KEY"),
    )

    _fs_tool_map = _BUILD_TOOL_MAP()  # see note at bottom — pull this out of
                                       # the function so it's built once at
                                       # import time, not on every chat msg
    _fs_schema = _AIEM_AGENT_TOOLS

    session_system = (
        _AIEM_AGENT_SYSTEM +
        f"\n\nSESSION FOCUS ({session_name}): {focus_prompt}\n"
        "Start immediately with the most relevant tools for this session's focus."
    )

    messages = [
        {"role": "system", "content": session_system},
        {"role": "user", "content": focus_prompt},
    ]

    trace = []
    _last_text = ""
    _context_budget = 60_000  # rough char budget for accumulated tool output
    _context_used = 0

    for _i in range(max_iterations):
        # --- retry wrapper around the model call ---
        resp = None
        for attempt in range(3):
            try:
                resp = _oai.chat.completions.create(
                    model="gpt-4o",
                    messages=messages,
                    tools=_fs_schema,
                    tool_choice="auto",
                    max_tokens=2000,
                    temperature=0.3,
                )
                break
            except Exception as e:
                if attempt == 2:
                    return _last_text, trace, f"OpenAI call failed after retries: {e}"
                time.sleep(1.5 * (attempt + 1))

        msg = resp.choices[0].message

        # Convert SDK message -> plain dict before storing. This is the
        # actual fix for bug #1.
        msg_dict = {"role": "assistant", "content": msg.content or None}
        if msg.tool_calls:
            msg_dict["tool_calls"] = [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {"name": tc.function.name, "arguments": tc.function.arguments},
                }
                for tc in msg.tool_calls
            ]
        messages.append(msg_dict)

        if msg.content:
            _last_text = msg.content

        if not msg.tool_calls:
            break  # model gave a final answer with no further tool use

        for tc in msg.tool_calls:
            fn = tc.function.name

            try:
                args = json.loads(tc.function.arguments or "{}")
            except json.JSONDecodeError as e:
                result = {"error": f"malformed arguments from model: {e}"}
                args = None

            if args is not None:
                if fn not in _fs_tool_map:
                    result = {"error": f"unknown tool: {fn}"}
                else:
                    try:
                        result = _fs_tool_map[fn](**args)
                    except Exception as e:
                        # THE KEY FIX: a tool blowing up no longer kills the
                        # whole session. The model sees the error and can
                        # adapt (try different args, a different tool, etc).
                        result = {"error": f"{fn} raised: {type(e).__name__}: {e}"}

            result_str = json.dumps(result, default=str)
            if len(result_str) > 6000:
                result_str = result_str[:6000] + "...[truncated]"

            _context_used += len(result_str)
            if _context_used > _context_budget:
                # Stop feeding full tool output once we're past budget;
                # keep going but tell the model output is being summarized.
                result_str = json.dumps({"note": "result omitted, context budget reached"})

            step = {
                "iteration": _i,
                "tool": fn,
                "args": args,
                "ok": "error" not in (result if isinstance(result, dict) else {}),
            }
            trace.append(step)
            if on_step:
                try:
                    on_step(step)
                except Exception:
                    pass  # never let a progress-callback failure kill the session

            messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": result_str,
            })

    return _last_text, trace, None

def classify_question_complexity(question: str) -> int:
    """
    Answers improvement-area #2: dynamic max_iterations instead of a flat 3.

    Flat 3 iterations is too low for anything that needs the agent to
    gather data AND validate it (e.g. "is quiet accumulation real?" needs
    mkt_quiet_accumulation -> mkt_test_signal -> mkt_validate_oos, which is
    already 3 tool calls before it's written a word of answer). Bump the
    budget based on cheap keyword heuristics; this doesn't need to be exact,
    it just needs to stop starving multi-step questions.
    """
    q = question.lower()

    deep_signals = ["backtest", "validate", "significan", "out-of-sample",
                     "oos", "is it real", "does it work", "prove", "p-value"]
    compare_signals = ["compare", "vs", "versus", "better than", "which of"]
    simple_signals = ["what is", "define", "explain", "how does"]

    if any(s in q for s in deep_signals):
        return 10
    if any(s in q for s in compare_signals):
        return 7
    if any(s in q for s in simple_signals) and len(q) < 80:
        return 4
    return 6  # default: more breathing room than your old flat 3

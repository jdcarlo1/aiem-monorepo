# =========================================
# 2. FEATURE ENGINE (REAL INDICATORS)
# =========================================

class FeatureEngine:

    def add_features(self, df):

        df["returns"] = df["close"].pct_change()
        df["volatility"] = df["returns"].rolling(10).std()

        df["momentum"] = df["close"] - df["close"].shift(5)

        df["volume_z"] = (
            (df["volume"] - df["volume"].rolling(20).mean()) /
            df["volume"].rolling(20).std()
        )

        return df


# =========================================
# 3. STRATEGY ENGINE (SIGNAL GENERATION)
# =========================================

class SignalEngine:

    def generate_signal(self, row):

        score = 0

        if row["momentum"] > 0:
            score += 0.4

        if row["volume_z"] > 1:
            score += 0.3

        if row["volatility"] < 0.02:
            score += 0.3

        return min(max(score, 0), 1)


# =========================================
# 4. BACKTEST ENGINE (REAL EVALUATION)
# =========================================

class BacktestEngine:

    def run_backtest(self, df, signal_engine):

        positions = []
        equity = 10000
        position = 0

        for i in range(20, len(df)):

            row = df.iloc[i]
            signal = signal_engine.generate_signal(row)

            # ENTRY RULE
            if signal > 0.7 and position == 0:
                position = equity / row["close"]
                entry_price = row["close"]

            # EXIT RULE
            elif signal < 0.3 and position > 0:
                equity = position * row["close"]
                position = 0

            positions.append(equity)

        return positions


# =========================================
# 5. PERFORMANCE METRICS (CRITICAL LEVEL 2)
# =========================================

class MetricsEngine:

    def analyze(self, equity_curve):

        returns = pd.Series(equity_curve).pct_change().dropna()

        sharpe = np.mean(returns) / (np.std(returns) + 1e-9)
        max_drawdown = (pd.Series(equity_curve).cummax() - equity_curve).max()

        win_rate = (returns > 0).mean()

        return {
            "sharpe_ratio": sharpe,
            "max_drawdown": max_drawdown,
            "win_rate": win_rate
        }


# =========================================
# 6. VALIDATION LAYER (ANTI-OVERFITTING)

# =========================================
# 7. MAIN AEIM LEVEL 2 SYSTEM
# =========================================

class AEIM_Level2:

    def __init__(self):
        self.data = MarketDataEngine()
        self.features = FeatureEngine()
        self.signal = SignalEngine()
        self.backtest = BacktestEngine()
        self.metrics = MetricsEngine()
        self.validate = ValidationEngine()

    def run(self, symbol):

        print(f"\n=== AEIM LEVEL 2 RUN: {symbol} ===")

        # DATA
        df = self.data.fetch_ohlcv(symbol)

        # FEATURES
        df = self.features.add_features(df)

        # BACKTEST
        equity_curve = self.backtest.run_backtest(df, self.signal)

        # METRICS
        results = self.metrics.analyze(equity_curve)

        print("\n--- PERFORMANCE ---")
        print(results)

        # VALIDATION
        validation_score = self.validate.walk_forward_test(df, self.signal)

        print("\n--- VALIDATION SCORE ---")
        print(validation_score)

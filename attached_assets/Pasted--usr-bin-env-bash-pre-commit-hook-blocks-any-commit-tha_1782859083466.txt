#!/usr/bin/env bash
#
# pre-commit hook: blocks any commit that touches precursor_signals.py or
# event_study_backtest.py unless verify_signals.py passes cleanly.
#
# INSTALL (run once in your Replit shell, from the project root):
#   cp pre-commit .git/hooks/pre-commit
#   chmod +x .git/hooks/pre-commit
#
# After that, this runs automatically on every `git commit` -- nobody,
# including the AI agent, has to remember to invoke it. If verification
# fails, the commit is rejected and nothing gets saved to history until
# it's fixed.
#
# NOTE ON LIMITS (read this so you know what this does and doesn't cover):
#   - This only protects files committed through git. If the agent edits
#     a file and never commits, or you're just running code live in the
#     Replit shell without committing, this hook never fires.
#   - `git commit --no-verify` skips this hook entirely. It's a flag
#     anyone (or any agent) can pass. Worth explicitly telling your
#     agent never to use --no-verify on this repo.
#   - It only catches what verify_signals.py actually tests. New
#     functions need new test cases added to verify_signals.py, or
#     they're invisible to this gate even though it says "passed."

set -e

CHANGED_FILES=$(git diff --cached --name-only)

NEEDS_CHECK=false
for f in $CHANGED_FILES; do
    if [[ "$f" == "precursor_signals.py" || "$f" == "event_study_backtest.py" || "$f" == "verify_signals.py" ]]; then
        NEEDS_CHECK=true
    fi
done

if [ "$NEEDS_CHECK" = false ]; then
    # none of the guarded files changed in this commit, nothing to check
    exit 0
fi

echo ""
echo "=================================================================="
echo "pre-commit hook: guarded file changed, running verify_signals.py"
echo "=================================================================="

python3 verify_signals.py
RESULT=$?

if [ $RESULT -ne 0 ]; then
    echo ""
    echo "=================================================================="
    echo "COMMIT BLOCKED: verify_signals.py failed."
    echo "Fix the failing checks before committing changes to"
    echo "precursor_signals.py / event_study_backtest.py."
    echo "=================================================================="
    exit 1
fi

echo ""
echo "verify_signals.py passed -- commit allowed."
exit 0

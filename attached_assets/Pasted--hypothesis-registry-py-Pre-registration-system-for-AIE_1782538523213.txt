"""
hypothesis_registry.py
-----------------------
Pre-registration system for AIEM's hypothesis testing.

PURPOSE
The single biggest threat to AIEM's win-rate claims is the multiple-comparisons
problem: if AIEM (or a human) tests 50 variations of a signal and reports the
best one, the "81% win rate" is largely an artifact of how many things were
tried, not a real edge.

This module forces every hypothesis to be:
  1. Registered with its exact parameters BEFORE it is tested
  2. Assigned a fixed, immutable train/test split at registration time
  3. Locked (read-only) the moment a result is recorded
  4. Counted, so a multiple-comparisons correction can always be computed

Nothing in this file can be edited retroactively to make a result look
better. If you need to change a hypothesis, you register a NEW one.

REQUIRES: a Postgres connection string in env var AIEM_DATABASE_URL.
This should point at AIEM's OWN schema/database, never production.
"""

import os
import json
import hashlib
import datetime as dt
from dataclasses import dataclass, field, asdict
from typing import Optional, List, Dict, Any

import psycopg2
import psycopg2.extras


DDL = """
CREATE TABLE IF NOT EXISTS hypothesis_registry (
    id SERIAL PRIMARY KEY,
    hypothesis_hash TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    description TEXT,
    parameters JSONB NOT NULL,
    train_start DATE NOT NULL,
    train_end DATE NOT NULL,
    test_start DATE NOT NULL,
    test_end DATE NOT NULL,
    registered_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    locked BOOLEAN NOT NULL DEFAULT FALSE,
    result JSONB,
    result_recorded_at TIMESTAMPTZ
);

-- Running count of ALL hypotheses ever registered, for multiple-comparisons
-- correction. This counter is append-only and never decremented.
CREATE TABLE IF NOT EXISTS hypothesis_counter (
    id INT PRIMARY KEY DEFAULT 1,
    total_registered INT NOT NULL DEFAULT 0,
    CHECK (id = 1)
);
INSERT INTO hypothesis_counter (id, total_registered)
VALUES (1, 0) ON CONFLICT (id) DO NOTHING;
"""


def _connect():
    url = os.environ.get("AIEM_DATABASE_URL")
    if not url:
        raise RuntimeError(
            "AIEM_DATABASE_URL is not set. Refusing to fall back to a "
            "production DB var (e.g. DATABASE_URL) — set this explicitly "
            "to AIEM's own database to keep isolation intact."
        )
    return psycopg2.connect(url)


def init_schema():
    with _connect() as conn:
        with conn.cursor() as cur:
            cur.execute(DDL)
        conn.commit()


@dataclass
class Hypothesis:
    name: str
    description: str
    parameters: Dict[str, Any]
    train_start: str  # ISO date
    train_end: str
    test_start: str
    test_end: str

    def hash(self) -> str:
        """Deterministic hash of the hypothesis definition. Two calls with
        identical name+parameters+window produce the same hash, which is
        what prevents silently re-registering the 'same' test with a new
        timestamp to dodge the multiple-comparisons count."""
        payload = json.dumps(
            {
                "name": self.name,
                "parameters": self.parameters,
                "train_start": self.train_start,
                "train_end": self.train_end,
                "test_start": self.test_start,
                "test_end": self.test_end,
            },
            sort_keys=True,
        )
        return hashlib.sha256(payload.encode()).hexdigest()


def register_hypothesis(h: Hypothesis) -> int:
    """Register a hypothesis BEFORE running any test against the test window.

    Raises if test_start is in the past relative to when this hypothesis's
    train window ends — i.e. you cannot register a hypothesis whose test
    period has already partly elapsed without flagging it, since that opens
    the door to peeking.
    """
    test_start_date = dt.date.fromisoformat(h.test_start)
    train_end_date = dt.date.fromisoformat(h.train_end)
    if test_start_date <= train_end_date:
        raise ValueError("test_start must be strictly after train_end")

    h_hash = h.hash()
    with _connect() as conn:
        with conn.cursor() as cur:
            cur.execute(
                "SELECT id, locked FROM hypothesis_registry WHERE hypothesis_hash = %s",
                (h_hash,),
            )
            existing = cur.fetchone()
            if existing:
                raise ValueError(
                    f"This exact hypothesis is already registered (id={existing[0]}, "
                    f"locked={existing[1]}). Register a genuinely new variant instead "
                    f"of re-registering the same one."
                )

            cur.execute(
                """
                INSERT INTO hypothesis_registry
                    (hypothesis_hash, name, description, parameters,
                     train_start, train_end, test_start, test_end)
                VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
                RETURNING id
                """,
                (
                    h_hash, h.name, h.description, json.dumps(h.parameters),
                    h.train_start, h.train_end, h.test_start, h.test_end,
                ),
            )
            new_id = cur.fetchone()[0]

            cur.execute(
                "UPDATE hypothesis_counter SET total_registered = total_registered + 1 WHERE id = 1"
            )
        conn.commit()
    return new_id


def record_result(hypothesis_id: int, result: Dict[str, Any]):
    """Record a test result and PERMANENTLY LOCK the hypothesis row.
    Once locked, no further updates to parameters or result are possible
    through this module (enforced in application logic; for true DB-level
    enforcement, also REVOKE UPDATE on this table for non-owner roles after
    insert, or use a trigger — see lock_trigger.sql below)."""
    with _connect() as conn:
        with conn.cursor() as cur:
            cur.execute(
                "SELECT locked FROM hypothesis_registry WHERE id = %s", (hypothesis_id,)
            )
            row = cur.fetchone()
            if row is None:
                raise ValueError(f"No hypothesis with id={hypothesis_id}")
            if row[0]:
                raise PermissionError(
                    f"Hypothesis {hypothesis_id} is already locked. Results cannot "
                    f"be overwritten. Register a new hypothesis if you need to retest."
                )
            cur.execute(
                """
                UPDATE hypothesis_registry
                SET result = %s, result_recorded_at = now(), locked = TRUE
                WHERE id = %s
                """,
                (json.dumps(result), hypothesis_id),
            )
        conn.commit()


def get_total_registered() -> int:
    """Total number of hypotheses EVER registered (tested or not).
    Use this as 'n' in a Bonferroni or Benjamini-Hochberg correction before
    trusting any single hypothesis's p-value or win rate."""
    with _connect() as conn:
        with conn.cursor() as cur:
            cur.execute("SELECT total_registered FROM hypothesis_counter WHERE id = 1")
            return cur.fetchone()[0]


def bonferroni_adjusted_alpha(alpha: float = 0.05) -> float:
    """Simple, conservative multiple-comparisons correction.
    If you've tested 50 hypotheses, alpha=0.05 becomes 0.001 — meaning a
    hypothesis needs a much stronger result to be trusted."""
    n = max(get_total_registered(), 1)
    return alpha / n


def list_locked_results() -> List[Dict[str, Any]]:
    """Return all locked (i.e. truthfully reported) hypothesis results,
    for audit / external verification by your separate review agent."""
    with _connect() as conn:
        with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(
                """
                SELECT id, name, parameters, train_start, train_end,
                       test_start, test_end, registered_at,
                       result, result_recorded_at
                FROM hypothesis_registry
                WHERE locked = TRUE
                ORDER BY registered_at ASC
                """
            )
            return [dict(r) for r in cur.fetchall()]


# Optional DB-level enforcement (run once, as an admin, against AIEM's schema).
# This makes "locked" rows literally un-updatable even if someone bypasses
# this Python module entirely.
LOCK_TRIGGER_SQL = """
CREATE OR REPLACE FUNCTION prevent_locked_update() RETURNS TRIGGER AS $$
BEGIN
    IF OLD.locked = TRUE THEN
        RAISE EXCEPTION 'Cannot modify a locked hypothesis (id=%)', OLD.id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS lock_enforcement ON hypothesis_registry;
CREATE TRIGGER lock_enforcement
    BEFORE UPDATE ON hypothesis_registry
    FOR EACH ROW
    EXECUTE FUNCTION prevent_locked_update();
"""


if __name__ == "__main__":
    init_schema()
    print("hypothesis_registry schema ready.")
    print(f"Total hypotheses registered so far: {get_total_registered()}")

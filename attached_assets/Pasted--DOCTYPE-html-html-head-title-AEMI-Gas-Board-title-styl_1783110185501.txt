<script>
async function runAEMI() {

  const risk = document.getElementById("risk").value;
  const strategy = document.getElementById("strategy").value;
  const minScore = parseInt(document.getElementById("minScore").value || "70");
  const tickers = document.getElementById("tickers").value.split(",");

  const res = await fetch("http://localhost:8000/aemi/signal", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      user_id: "demo",
      risk_level: risk,
      strategy: strategy,
      min_score: minScore,
      watchlist_only: true,
      tickers: tickers
    })
  });

  const data = await res.json();

  const container = document.getElementById("results");
  container.innerHTML = "";

  data.signals.forEach(s => {

    const div = document.createElement("div");
    div.className = "card";

    div.innerHTML = `
      <div class="label">${s.strategy} • ${s.regime}</div>
      <div class="score ${s.signal === "bullish" ? "bullish" : "bearish"}">
        ${s.ticker} ${s.score}
      </div>
      <div>Probability: ${(s.probability * 100).toFixed(1)}%</div>
      <div class="meta">
        Risk: ${s.risk_level} | Signal: ${s.signal}
      </div>
    `;

    container.appendChild(div);
  });
}
</script>

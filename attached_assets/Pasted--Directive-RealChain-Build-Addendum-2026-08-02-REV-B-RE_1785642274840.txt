**C1.** Add these three helpers immediately above the `_sinc` block (module level or `_execute_job` body — match the surrounding indentation):

```python
_LIQ_IV_MAX = 3.0
_MIN_DTE = 5

def _pick_expiry(chain, min_dte=_MIN_DTE):
    """Nearest expiration with dte >= min_dte, taken from the chain itself."""
    exps = sorted({
        (c.get("dte"), c.get("expiration_date"))
        for c in (chain or [])
        if c.get("expiration_date") and c.get("dte") is not None and c.get("dte") >= min_dte
    })
    return exps[0][1] if exps else None

def _liquid_chain(chain, typ, expiry):
    out = []
    for c in chain or []:
        if c.get("contract_type") != typ:
            continue
        if expiry and c.get("expiration_date") != expiry:
            continue
        b, a = c.get("bid"), c.get("ask")
        iv, d = c.get("implied_volatility"), c.get("delta")
        if not b or not a or b <= 0 or a <= 0:
            continue
        if iv is None or iv <= 0 or iv > _LIQ_IV_MAX:
            continue
        if d is None or abs(d) == 0.0:
            continue
        out.append(c)
    return out

def _pick_by_delta(cands, target):
    return min(cands, key=lambda c: abs(abs(c.get("delta") or 0.0) - target))
```

`_pick_expiry` exists so no pre-existing expiration variable is required. If an expiration variable is already in scope at 1643 and disagrees with `_pick_expiry(options_chain)`, report both values and stop — do not silently pick one.

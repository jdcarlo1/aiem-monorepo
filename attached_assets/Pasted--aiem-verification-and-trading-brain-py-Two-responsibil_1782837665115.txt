# ── Core staleness checks ─────────────────────────────────────────────────────

def is_stale_gap(ticker: str, scan_ts: datetime) -> tuple[bool, str | None]:
    """Check 1 — did the gap happen yesterday or earlier?"""
    gap_open        = get_gap_open_price(ticker)
    current_price   = get_current_price(ticker)
    gap_origin_date = get_gap_date(ticker)
    today           = scan_ts.date()

    if gap_origin_date < today:
        return True, "GAP_IS_YESTERDAY"

    if gap_open and gap_open > 0:
        extension = (current_price - gap_open) / gap_open
        if extension > 0.30:
            return True, f"EXTENDED_FROM_GAP ({extension:.1%})"

    return False, None


def catalyst_conviction_penalty(ticker: str, scan_ts: datetime) -> int:
    """Check 2 — penalise stale catalysts."""
    try:
        catalyst_ts = get_catalyst_timestamp(ticker)
    except Exception:
        logger.warning("[%s] No catalyst timestamp — applying max penalty.", ticker)
        return -4

    if catalyst_ts.tzinfo is None:
        catalyst_ts = catalyst_ts.replace(tzinfo=timezone.utc)
    if scan_ts.tzinfo is None:
        scan_ts = scan_ts.replace(tzinfo=timezone.utc)

    age_hours = (scan_ts - catalyst_ts).total_seconds() / 3600

    if age_hours > 48:
        return -4
    if age_hours > 24:
        return -2
    return 0


def get_move_day(ticker: str, scan_date: date) -> int:
    """Check 3 — how many sessions old is this move?"""
    gap_origin_date = get_gap_date(ticker)
    delta = (scan_date - gap_origin_date).days
    return max(1, delta + 1)


def vwap_day2_tag(ticker: str, move_day: int) -> str | None:
    """Check 4 — day-2 exhaustion above VWAP on fading volume."""
    if move_day < 2:
        return None
    current_price = get_current_price(ticker)
    vwap          = get_vwap(ticker)
    vol_trend     = get_volume_trend(ticker)
    if vwap and vwap > 0:
        vwap_pos = (current_price - vwap) / vwap
        if vol_trend == "decreasing" and vwap_pos > 0.05:
            return "DAY2_EXTENDED_ABOVE_VWAP"
    return None

def apply_wall_street_pattern(ticker: str, signal: dict) -> dict:
    """
    Applies pattern-library rules on top of the base staleness filter.
    Adds pattern tags and further adjusts conviction.
    """
    tags       = signal.get("tags", [])
    conviction = signal.get("final_conviction", 0)
    notes      = []

    move_day = signal.get("move_day", 1)

    # Pattern: PIPE fade
    try:
        pipe_price    = signal.get("pipe_price")
        current_price = get_current_price(ticker)
        if pipe_price and current_price > pipe_price * 1.05:
            tags.append("PATTERN_PIPE_FADE")
            conviction -= 10
            notes.append(f"Trading {((current_price/pipe_price)-1):.1%} above PIPE price {pipe_price}; institutional seller in profit.")
    except Exception:
        pass

    # Pattern: sympathy play (no own catalyst)
    if "CATALYST_PENALTY(-4)" in tags or signal.get("catalyst_age_hours", 0) > 48:
        tags.append("PATTERN_SYMPATHY_PLAY")
        conviction -= 5
        notes.append("No fresh own catalyst — sympathy move only.")

    # Pattern: day-2 distribution
    if move_day >= 2 and "DAY2_EXTENDED_ABOVE_VWAP" in tags:
        tags.append("PATTERN_DAY2_DISTRIBUTION")
        conviction -= 5
        notes.append("Day-2 distribution: price above VWAP, volume declining.")

    signal["final_conviction"] = max(0, conviction)
    signal["tags"]             = list(set(tags))
    signal["ws_notes"]         = notes
    return signal


def evaluate_signal(
    ticker: str,
    base_conviction: int,
    scan_timestamp: datetime,
    extra: dict | None = None,
) -> dict:
    """
    Master evaluation: staleness filter + Wall Street pattern library.
    extra dict can carry:  pipe_price, catalyst_age_hours, etc.
    """
    extra     = extra or {}
    tags      = []
    scan_date = scan_timestamp.date()

    # ── Staleness checks ──────────────────────────────────────────────────
    stale, stale_reason = is_stale_gap(ticker, scan_timestamp)
    if stale:
        return {
            "ticker":           ticker,
            "action":           "SKIP",
            "final_conviction": 0,
            "move_day":         None,
            "tags":             [stale_reason],
            "reason":           f"Stale gap: {stale_reason}",
            "ws_notes":         [WALL_STREET_RULES],
        }

    penalty = catalyst_conviction_penalty(ticker, scan_timestamp)
    adjusted = base_conviction + penalty
    if penalty < 0:
        tags.append(f"CATALYST_PENALTY({penalty})")

    move_day = get_move_day(ticker, scan_date)
    if move_day >= 2:
        tags.append(f"MOVE_DAY_{move_day}")

    vwap_tag = vwap_day2_tag(ticker, move_day)
    if vwap_tag:
        tags.append(vwap_tag)
        adjusted -= 3

    CONVICTION_THRESHOLD = 70

    signal = {
        "ticker":           ticker,
        "action":           "PASS" if adjusted >= CONVICTION_THRESHOLD else "SKIP",
        "final_conviction": max(0, adjusted),
        "move_day":         move_day,
        "tags":             tags,
        "reason":           (
            f"Conviction {adjusted} cleared threshold {CONVICTION_THRESHOLD}"
            if adjusted >= CONVICTION_THRESHOLD
            else f"Conviction {adjusted} below threshold {CONVICTION_THRESHOLD}"
        ),
        "ws_notes":         [],
        **extra,
    }

    # ── Wall Street pattern layer ─────────────────────────────────────────
    signal = apply_wall_street_pattern(ticker, signal)

    logger.info(
        "[%s] %s | conviction %d→%d | tags %s",
        ticker, signal["action"], base_conviction,
        signal["final_conviction"], signal["tags"],
    )
    return signal

def _hmac_sign(payload: str) -> str:
    """Return hex HMAC-SHA256 of payload using AIEM_HMAC_SECRET."""
    return hmac.new(HMAC_SECRET, payload.encode(), hashlib.sha256).hexdigest()


def issue_challenge(question_id: int) -> dict:
    """
    Generate a signed challenge for a given question.
    Returns a JSON-serialisable dict that you paste to AIEM.

    {
        "question_id": 3,
        "nonce":       "<uuid4>",
        "issued_at":   1234567890,
        "question":    "...",
        "sig":         "<hmac-hex>"      ← covers nonce + question_id + issued_at
    }
    """
    if question_id not in VERIFICATION_QUESTIONS:
        raise ValueError(f"Unknown question_id {question_id}. Valid: {list(VERIFICATION_QUESTIONS)}")

    nonce     = str(uuid.uuid4())
    issued_at = int(time.time())
    payload   = f"{nonce}:{question_id}:{issued_at}"
    sig       = _hmac_sign(payload)

    challenge = {
        "question_id": question_id,
        "nonce":       nonce,
        "issued_at":   issued_at,
        "question":    VERIFICATION_QUESTIONS[question_id]["question"],
        "sig":         sig,
    }
    logger.info("Challenge issued for Q%d  nonce=%s", question_id, nonce)
    return challenge


def verify_response(challenge: dict, aiem_answer: str) -> dict:
    """
    Verify that:
      (a) the challenge signature is authentic (wasn't forged)
      (b) the challenge hasn't expired
      (c) the answer contains the expected keywords

    Returns:
    {
        "valid":        bool,
        "verdict":      str,
        "missing_keys": list[str],
        "explanation":  str
    }
    """
    question_id = challenge.get("question_id")
    nonce       = challenge.get("nonce")
    issued_at   = challenge.get("issued_at")
    received_sig = challenge.get("sig", "")

    # ── (a) Verify signature ──────────────────────────────────────────────
    expected_payload = f"{nonce}:{question_id}:{issued_at}"
    expected_sig     = _hmac_sign(expected_payload)

    if not hmac.compare_digest(expected_sig, received_sig):
        return {
            "valid":        False,
            "verdict":      "INVALID_SIGNATURE — challenge was forged or tampered with.",
            "missing_keys": [],
            "explanation":  VERIFICATION_QUESTIONS.get(question_id, {}).get("explanation", ""),
        }

    # ── (b) Check TTL ────────────────────────────────────────────────────
    age = int(time.time()) - issued_at
    if age > CHALLENGE_TTL_SECONDS:
        return {
            "valid":        False,
            "verdict":      f"CHALLENGE_EXPIRED — issued {age}s ago (TTL {CHALLENGE_TTL_SECONDS}s).",
            "missing_keys": [],
            "explanation":  VERIFICATION_QUESTIONS.get(question_id, {}).get("explanation", ""),
        }

    # ── (c) Keyword check ────────────────────────────────────────────────
    q_meta   = VERIFICATION_QUESTIONS[question_id]
    keywords = q_meta["expected_keywords"]
    lower_ans = aiem_answer.lower()
    missing  = [kw for kw in keywords if kw.lower() not in lower_ans]

    if missing:
        verdict = f"PARTIAL — AIEM answer is missing expected content: {missing}"
        valid   = False
    else:
        verdict = "VERIFIED ✓ — AIEM answer contains all expected content."
        valid   = True

    return {
        "valid":        valid,
        "verdict":      verdict,
        "missing_keys": missing,
        "explanation":  q_meta["explanation"],
    }


def run_all_challenges() -> list[dict]:
    """
    Issue all 6 challenges in sequence and return their challenge dicts.
    Paste each one to AIEM, then call verify_response() with its answer.
    """
    challenges = []
    for qid in sorted(VERIFICATION_QUESTIONS):
        ch = issue_challenge(qid)
        challenges.append(ch)
        print(f"\n{'─'*70}")
        print(f"  CHALLENGE Q{qid}  (nonce={ch['nonce'][:8]}…)")
        print(f"  Question: {ch['question']}")
        print(f"  Sig:      {ch['sig'][:16]}…")
    return challenges

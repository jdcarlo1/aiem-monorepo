# ─────────────────────────────────────────────────────────────
# DATABASE POOL — standalone, not shared with Flask
# ─────────────────────────────────────────────────────────────
_AIEM_POOL = None

def _init_pool():
    global _AIEM_POOL
    if _AIEM_POOL is None:
        _AIEM_POOL = psycopg2.pool.ThreadedConnectionPool(
            minconn=1,
            maxconn=5,
            dsn=DATABASE_URL
        )
        log.info("DB pool initialized (min=1 max=5)")

def _get_conn():
    global _AIEM_POOL
    if _AIEM_POOL is None:
        _init_pool()
    return _AIEM_POOL.getconn()

def _put_conn(conn):
    global _AIEM_POOL
    if _AIEM_POOL and conn:
        _AIEM_POOL.putconn(conn)

# POLYGON HELPERS — standalone wrappers
# ─────────────────────────────────────────────────────────────
POLYGON_BASE = "https://api.polygon.io"

def _aiem_bulk_snapshot() -> list:
    """
    Pull ALL stocks snapshot from Polygon in one call.
    Returns list of ticker dicts with premarket price, volume, gap.
    This is the 8,000+ stock universe scan — runs in ~3 seconds.
    """
    try:
        url = f"{POLYGON_BASE}/v2/snapshot/locale/us/markets/stocks/tickers"
        params = {
            'apiKey':         POLYGON_API_KEY,
            'include_otc':    'false',
        }
        r = requests.get(url, params=params, timeout=30)
        r.raise_for_status()
        data = r.json()
        tickers = data.get('tickers', [])
        log.info(f"Bulk snapshot returned {len(tickers)} tickers")
        return tickers
    except Exception as e:
        log.error(f"Bulk snapshot error: {e}")
        return []

def _aiem_get_snapshot(ticker: str) -> dict:
    """Single ticker snapshot — for deep analysis of finalists."""
    try:
        url = f"{POLYGON_BASE}/v2/snapshot/locale/us/markets/stocks/tickers/{ticker}"
        r = requests.get(url, params={'apiKey': POLYGON_API_KEY}, timeout=10)
        r.raise_for_status()
        data = r.json()
        return data.get('ticker', {})
    except Exception as e:
        log.error(f"Snapshot error {ticker}: {e}")
        return {}

def _aiem_get_ohlcv(ticker: str, days: int = 5) -> list:
    """Get recent OHLCV bars for pattern analysis."""
    try:
        end   = date.today()
        start = end - timedelta(days=days + 5)
        url   = f"{POLYGON_BASE}/v2/aggs/ticker/{ticker}/range/1/day/{start}/{end}"
        r = requests.get(url, params={'apiKey': POLYGON_API_KEY, 'adjusted': 'true'}, timeout=10)
        r.raise_for_status()
        return r.json().get('results', [])
    except Exception as e:
        log.error(f"OHLCV error {ticker}: {e}")
        return []

def _aiem_get_grouped_daily(for_date: date = None) -> list:
    """Get all stocks OHLCV for a specific date — used for missed runner analysis."""
    try:
        d   = for_date or date.today()
        url = f"{POLYGON_BASE}/v2/aggs/grouped/locale/us/market/stocks/{d}"
        r   = requests.get(url, params={'apiKey': POLYGON_API_KEY, 'adjusted': 'true'}, timeout=30)
        r.raise_for_status()
        results = r.json().get('results', [])
        log.info(f"Grouped daily for {d}: {len(results)} stocks")
        return results
    except Exception as e:
        log.error(f"Grouped daily error: {e}")
        return []

def _aiem_get_ticker_details(ticker: str) -> dict:
    """Get float, market cap, shares outstanding from Polygon."""
    try:
        url = f"{POLYGON_BASE}/v3/reference/tickers/{ticker}"
        r   = requests.get(url, params={'apiKey': POLYGON_API_KEY}, timeout=10)
        r.raise_for_status()
        return r.json().get('results', {})
    except Exception as e:
        log.error(f"Ticker details error {ticker}: {e}")
        return {}

def _aiem_get_news(ticker: str) -> list:
    """Check for recent news/catalyst — information lag detection."""
    try:
        url = f"{POLYGON_BASE}/v2/reference/news"
        params = {
            'apiKey':        POLYGON_API_KEY,
            'ticker':        ticker,
            'published_utc.gte': (datetime.now() - timedelta(hours=12)).strftime('%Y-%m-%dT%H:%M:%SZ'),
            'limit':         5
        }
        r = requests.get(url, params=params, timeout=10)
        r.raise_for_status()
        return r.json().get('results', [])
    except Exception as e:
        log.error(f"News error {ticker}: {e}")
        return []

# ─────────────────────────────────────────────────────────────
# SIGNAL TRUST WEIGHTS
# ─────────────────────────────────────────────────────────────
def _load_trust_weights(conn) -> dict:
    """Load AIEM's learned signal weights from DB."""
    try:
        cur = conn.cursor()
        cur.execute("""
            SELECT signal_name, trust_weight, rolling_win_rate, n_outcomes_observed
            FROM signal_trust_weights
            WHERE context_bucket = 'AIEM_MICROCAP'
            ORDER BY trust_weight DESC
        """)
        rows = cur.fetchall()
        weights = {}
        for name, trust, win_rate, n in rows:
            weights[name] = {
                'trust':    float(trust    or 1.0),
                'win_rate': float(win_rate or 0.5),
                'n':        n or 0
            }
        log.info(f"Loaded {len(weights)} signal trust weights")
        return weights
    except Exception as e:
        log.error(f"Trust weight load error: {e}")
        return {}

# ─────────────────────────────────────────────────────────────
# PRICING INEFFICIENCY DETECTION
# Six types AIEM hunts specifically in microcap universe
# ─────────────────────────────────────────────────────────────
def _detect_pricing_inefficiencies(ticker: str, snap: dict, details: dict, news: list) -> dict:
    """
    Detect exploitable pricing inefficiencies specific to microcaps.
    Returns dict of detected inefficiencies with scores.
    """
    inefficiencies = {}

    try:
        day        = snap.get('day', {})
        prev_day   = snap.get('prevDay', {})
        last_quote = snap.get('lastQuote', {})
        last_trade = snap.get('lastTrade', {})

        current_price = last_trade.get('p') or day.get('c') or 0
        prev_close    = prev_day.get('c') or 0
        volume        = day.get('v') or 0
        avg_volume    = snap.get('day', {}).get('vw') or 1
        bid           = last_quote.get('P') or 0
        ask           = last_quote.get('p') or 0
        float_shares  = details.get('share_class_shares_outstanding') or 0
        short_int     = details.get('short_interest') or 0

        gap_pct = ((current_price - prev_close) / prev_close * 100) if prev_close else 0

        # ── 1. INFORMATION LAG
        # News hit in last 2 hours but price hasn't fully reacted yet
        if news:
            newest_news_age = (datetime.now() - datetime.fromisoformat(
                news[0].get('published_utc', '').replace('Z', '+00:00')
            )).seconds / 3600 if news[0].get('published_utc') else 99
            if newest_news_age < 2 and gap_pct < 15:
                inefficiencies['information_lag'] = {
                    'score':       18,
                    'description': f"News {newest_news_age:.1f}hrs old, price only moved {gap_pct:.1f}% — lag opportunity"
                }

        # ── 2. BID/ASK SPREAD TIGHTENING
        # Spread narrowing premarket = institutional accumulation signal
        if bid > 0 and ask > 0:
            spread_pct = ((ask - bid) / bid * 100)
            if spread_pct < 0.3:
                inefficiencies['spread_tightening'] = {
                    'score':       15,
                    'description': f"Spread only {spread_pct:.2f}% — unusually tight for microcap, smart money accumulating"
                }
            elif spread_pct < 0.8:
                inefficiencies['spread_tightening'] = {
                    'score':       8,
                    'description': f"Spread {spread_pct:.2f}% — tightening, watch closely"
                }

        # ── 3. SHORT SQUEEZE GEOMETRY
        # Mathematical squeeze setup — when the numbers make it inevitable
        if float_shares > 0 and short_int > 0:
            short_pct    = (short_int / float_shares * 100)
            vol_vs_float = (volume / float_shares * 100) if float_shares > 0 else 0
            if short_pct >= 20 and vol_vs_float >= 10:
                inefficiencies['squeeze_geometry'] = {
                    'score':       22,
                    'description': f"SQUEEZE MATH: {short_pct:.1f}% short, {vol_vs_float:.1f}% float traded — squeeze inevitable"
                }
            elif short_pct >= 15:
                inefficiencies['squeeze_geometry'] = {
                    'score':       14,
                    'description': f"Short {short_pct:.1f}% of float — squeeze fuel building"
                }

        # ── 4. CATALYST MISPRICING
        # Significant catalyst but stock hasn't repriced correctly yet
        if news and gap_pct < 20:
            keywords = ['fda', 'approval', 'contract', 'partnership', 'merger',
                       'acquisition', 'revenue', 'milestone', 'grant', 'award']
            titles   = ' '.join([n.get('title', '').lower() for n in news])
            hits     = [k for k in keywords if k in titles]
            if hits:
                inefficiencies['catalyst_mispricing'] = {
                    'score':       20,
                    'description': f"Catalyst detected ({', '.join(hits[:2])}) but only {gap_pct:.1f}% move — underpriced"
                }

        # ── 5. HALT PATTERN
        # Stocks that halted premarket tend to have explosive resumes
        # Detect via extreme gap with low current volume (just resumed)
        if gap_pct >= 20 and volume < 100_000:
            inefficiencies['halt_pattern'] = {
                'score':       16,
                'description': f"Gap {gap_pct:.1f}% with only {volume:,} volume — possible halt resume, explosive potential"
                }

        # ── 6. DARK POOL DIVERGENCE
        # When official volume is low but price is moving — dark pool accumulation
        vwap      = day.get('vw') or current_price
        price_dev = abs(current_price - vwap) / vwap * 100 if vwap else 0
        if price_dev > 3 and volume < 50_000:
            inefficiencies['dark_pool_divergence'] = {
                'score':       12,
                'description': f"Price {price_dev:.1f}% from VWAP with low public volume — dark pool activity suspected"
            }

    except Exception as e:
        log.error(f"Inefficiency detection error {ticker}: {e}")

    return inefficiencies


# ─────────────────────────────────────────────────────────────
# AIEM MICROCAP SCORER
# Self-weighted, trust-adjusted, inefficiency-aware
# ─────────────────────────────────────────────────────────────
def _aiem_score_microcap(ticker: str, snap: dict, details: dict,
                          news: list, trust_weights: dict) -> tuple:
    """
    AIEM's autonomous microcap scoring engine.
    Combines trust-weighted signals + pricing inefficiency detection.
    Returns: (confidence_score, signal_basis, reasoning, predicted_move)

    AIEM starts with seed signals but over time discovers its own
    via the nightly learning loop. Trust weights shift what matters.
    """
    raw_score = 0.0
    max_score = 0.0
    signals   = {}
    reasoning = []

    try:
        day        = snap.get('day', {})
        prev_day   = snap.get('prevDay', {})
        last_trade = snap.get('lastTrade', {})

        price      = last_trade.get('p') or day.get('c') or 0
        prev_close = prev_day.get('c') or price or 1
        volume     = day.get('v') or 0
        avg_vol    = prev_day.get('v') or 1

        float_sh   = details.get('share_class_shares_outstanding') or 999_000_000
        mkt_cap    = details.get('market_cap') or 999_000_000_000

        gap_pct    = ((price - prev_close) / prev_close * 100) if prev_close else 0
        vol_ratio  = volume / avg_vol if avg_vol > 0 else 0

        def add_signal(name, base_weight, condition, desc):
            nonlocal raw_score, max_score
            t = trust_weights.get(name, {}).get('trust', 1.0)
            w = base_weight * t
            max_score += w
            if condition:
                signals[name] = round(w, 3)
                raw_score    += w
                if desc:
                    reasoning.append(desc)

        # ── Core microcap signals
        add_signal('gap_explosive',      20, gap_pct >= 20,  f"Explosive gap +{gap_pct:.1f}%")
        add_signal('gap_large',          15, 10 <= gap_pct < 20, f"Large gap +{gap_pct:.1f}%")
        add_signal('gap_moderate',       10, 5  <= gap_pct < 10, f"Moderate gap +{gap_pct:.1f}%")
        add_signal('gap_small',           5, 2  <= gap_pct < 5,  f"Small gap +{gap_pct:.1f}%")

        add_signal('volume_extreme',     22, vol_ratio >= 5,  f"Volume {vol_ratio:.1f}x avg — extreme")
        add_signal('volume_high',        16, 3 <= vol_ratio < 5, f"Volume {vol_ratio:.1f}x avg — high")
        add_signal('volume_elevated',     8, 1.5 <= vol_ratio < 3, f"Volume {vol_ratio:.1f}x avg")

        add_signal('float_micro',        20, float_sh < 5_000_000,  f"Micro float {float_sh/1e6:.1f}M — explosive")
        add_signal('float_low',          14, 5_000_000 <= float_sh < 10_000_000, f"Low float {float_sh/1e6:.1f}M")
        add_signal('float_medium',        7, 10_000_000 <= float_sh < 20_000_000, f"Float {float_sh/1e6:.1f}M")

        add_signal('mktcap_nano',        18, mkt_cap < 50_000_000,   f"Nano cap ${mkt_cap/1e6:.0f}M — no institution")
        add_signal('mktcap_micro',       12, 50_000_000 <= mkt_cap < 150_000_000, f"Micro cap ${mkt_cap/1e6:.0f}M")
        add_signal('mktcap_small',        6, 150_000_000 <= mkt_cap < 500_000_000, f"Small cap ${mkt_cap/1e6:.0f}M")

        add_signal('price_sweet_spot',    8, 1.0 <= price <= 10.0, f"Price ${price:.2f} in breakout sweet spot")
        add_signal('catalyst_present',   15, len(news) > 0, f"Catalyst: {news[0].get('title','')[:50] if news else ''}")

        # Gap + Volume combo — strongest single setup in microcaps
        add_signal('gap_vol_combo',      25, gap_pct >= 10 and vol_ratio >= 3,
                   f"POWER COMBO: gap {gap_pct:.1f}% + vol {vol_ratio:.1f}x")

        # ── Add pricing inefficiency scores
        inefficiencies = _detect_pricing_inefficiencies(ticker, snap, details, news)
        for ineff_name, ineff_data in inefficiencies.items():
            score = ineff_data['score']
            desc  = ineff_data['description']
            t     = trust_weights.get(ineff_name, {}).get('trust', 1.0)
            w     = score * t
            max_score += w
            raw_score += w
            signals[ineff_name] = round(w, 3)
            reasoning.append(desc)

    except Exception as e:
        log.error(f"Score error {ticker}: {e}")

    # Normalize to 0-100
    confidence = (raw_score / max_score * 100) if max_score > 0 else 0
    confidence = min(100.0, round(confidence, 1))

    if confidence >= 85:
        predicted_move = "VERY HIGH CONVICTION — strong breakout expected"
    elif confidence >= 72:
        predicted_move = "HIGH CONVICTION — breakout setup confirmed"
    elif confidence >= 58:
        predicted_move = "MODERATE — watch open behavior closely"
    else:
        predicted_move = "LOW — monitor only, no action"

    signal_basis   = ", ".join(signals.keys())
    reasoning_text = " | ".join(reasoning) if reasoning else "Insufficient signals"

    return confidence, signal_basis, reasoning_text, predicted_move

# FUNCTION 1: PREMARKET SCAN
# Runs every 15 min 7:00–9:30 AM ET
# ─────────────────────────────────────────────────────────────
def aiem_premarket_scan():
    """
    AIEM's morning hunt. Scans entire market, filters to microcap universe,
    deep-scores finalists, writes top picks to aiem_predictions.
    Completely standalone — runs whether Flask is up or down.
    """
    now_et = datetime.now(ET)
    log.info(f"=== PREMARKET SCAN START {now_et.strftime('%H:%M ET')} ===")

    conn = None
    try:
        conn = _get_conn()

        # ── Step 1: Bulk scan entire market (8,000+ stocks, ~3 seconds)
        all_tickers = _aiem_bulk_snapshot()
        if not all_tickers:
            log.warning("Bulk snapshot returned nothing — Polygon may be down")
            return

        # ── Step 2: Filter to microcap universe
        candidates = []
        for t in all_tickers:
            try:
                day      = t.get('day', {})
                prev_day = t.get('prevDay', {})
                ticker   = t.get('ticker', '')

                if not ticker or len(ticker) > 5:
                    continue

                price     = t.get('lastTrade', {}).get('p') or day.get('c') or 0
                prev_cl   = prev_day.get('c') or price or 1
                volume    = day.get('v') or 0
                gap_pct   = ((price - prev_cl) / prev_cl * 100) if prev_cl else 0

                # Apply microcap filters
                if price < MIN_PRICE or price > MAX_PRICE:
                    continue
                if gap_pct < MIN_GAP_PCT:
                    continue
                if volume < MIN_PREMARKET_VOLUME:
                    continue

                candidates.append({'ticker': ticker, 'snap': t, 'gap': gap_pct, 'volume': volume})

            except Exception:
                continue

        log.info(f"Universe filtered: {len(all_tickers)} → {len(candidates)} microcap candidates")

        if not candidates:
            log.warning("No microcap candidates passed filters")
            return

        # Sort by gap * volume ratio — strongest movers first
        candidates.sort(key=lambda x: x['gap'] * (x['volume'] / 100000), reverse=True)
        top_candidates = candidates[:50]  # Deep score top 50

        # ── Step 3: Load trust weights so AIEM uses its learned knowledge
        trust_weights = _load_trust_weights(conn)

        # ── Step 4: Deep score each finalist
        scored = []
        for c in top_candidates:
            ticker = c['ticker']
            try:
                # Get float, market cap, details
                details = _aiem_get_ticker_details(ticker)

                # Apply hard market cap filter now that we have it
                mkt_cap = details.get('market_cap') or 0
                if mkt_cap > MAX_MARKET_CAP and mkt_cap > 0:
                    continue

                float_sh = details.get('share_class_shares_outstanding') or 0
                if float_sh > MAX_FLOAT_SHARES and float_sh > 0:
                    continue

                # Check for catalyst/news
                news = _aiem_get_news(ticker)

                # AIEM scores this ticker
                conf, sig_basis, reasoning, predicted = _aiem_score_microcap(
                    ticker, c['snap'], details, news, trust_weights
                )

                scored.append({
                    'ticker':    ticker,
                    'conf':      conf,
                    'sig_basis': sig_basis,
                    'reasoning': reasoning,
                    'predicted': predicted,
                    'gap':       c['gap'],
                    'volume':    c['volume'],
                })

                log.info(f"  {ticker}: conf={conf:.1f} gap={c['gap']:.1f}% vol={c['volume']:,}")

                # Rate limit — be nice to Polygon
                time.sleep(0.1)

            except Exception as e:
                log.error(f"Deep score error {ticker}: {e}")
                continue

        if not scored:
            log.warning("No tickers survived deep scoring")
            return

        # Sort by confidence
        scored.sort(key=lambda x: x['conf'], reverse=True)
        top_picks = scored[:10]

        # ── Step 5: Write to aiem_predictions
        today = date.today()
        cur   = conn.cursor()
        cur.execute("DELETE FROM aiem_predictions WHERE prediction_date = %s", (today,))

        for rank, pick in enumerate(top_picks, 1):
            cur.execute("""
                INSERT INTO aiem_predictions
                    (prediction_date, ticker, rank, confidence_score,
                     signal_basis, reasoning, predicted_move, created_at)
                VALUES (%s, %s, %s, %s, %s, %s, %s, NOW())
            """, (
                today,
                pick['ticker'],
                rank,
                pick['conf'],
                pick['sig_basis'],
                pick['reasoning'],
                pick['predicted'],
            ))

        conn.commit()
        log.info(f"✅ Wrote {len(top_picks)} predictions for {today}")
        for p in top_picks[:5]:
            log.info(f"  #{p['ticker']} conf={p['conf']:.1f} — {p['reasoning'][:80]}")

    except Exception as e:
        log.error(f"premarket_scan error: {e}")
        if conn:
            conn.rollback()
    finally:
        _put_conn(conn)

# FUNCTION 2: OPEN WATCHER + AUTONOMOUS ALERTS
# Runs every 5 min 9:30–10:30 AM ET
# AIEM decides when to fire — not a fixed schedule
# ─────────────────────────────────────────────────────────────
def aiem_open_watcher():
    """
    Monitors AIEM's picks at open. Re-scores with live data.
    Fires SMS the moment confidence crosses threshold.
    AIEM chooses the alert time — could be 9:31, could be 10:14.
    """
    now_et  = datetime.now(ET)
    hour    = now_et.hour
    minute  = now_et.minute

    if not ((hour == 9 and minute >= 30) or hour == 10):
        return

    conn = None
    try:
        conn  = _get_conn()
        cur   = conn.cursor()
        today = date.today()

        # Load today's picks
        cur.execute("""
            SELECT ticker, rank, confidence_score, signal_basis, reasoning, predicted_move
            FROM aiem_predictions
            WHERE prediction_date = %s
            ORDER BY rank ASC
        """, (today,))
        picks = cur.fetchall()

        if not picks:
            log.info("No predictions for today — skipping open watch")
            return

        # Who already got an alert today?
        cur.execute("""
            SELECT ticker FROM signal_fire_log
            WHERE DATE(fired_at) = %s AND signal_name = 'AIEM_ALERT'
        """, (today,))
        alerted = set(row[0] for row in cur.fetchall())

        trust_weights = _load_trust_weights(conn)
        mins_since_open = (hour - 9) * 60 + max(0, minute - 30)

        for pick in picks:
            ticker, rank, base_conf, sig_basis, reasoning, predicted = pick

            if ticker in alerted:
                continue

            try:
                # Re-fetch live snapshot
                live_snap    = _aiem_get_snapshot(ticker)
                live_details = _aiem_get_ticker_details(ticker)
                live_news    = _aiem_get_news(ticker)

                live_conf, live_sigs, live_reason, live_predicted = _aiem_score_microcap(
                    ticker, live_snap, live_details, live_news, trust_weights
                )

                # Blend: premarket score + live open behavior
                # Weight live score more heavily as time passes open
                live_weight = min(0.8, 0.4 + (mins_since_open / 100))
                pre_weight  = 1.0 - live_weight
                blended     = (base_conf * pre_weight) + (live_conf * live_weight)

                log.info(f"{ticker}: blended={blended:.1f} "
                         f"(pre={base_conf:.1f} live={live_conf:.1f}) "
                         f"at {now_et.strftime('%H:%M')}")

                # AIEM decides: am I confident enough RIGHT NOW?
                if blended >= CONFIDENCE_ALERT_THRESHOLD:
                    time_label = f"{mins_since_open}min after open" if mins_since_open > 0 else "AT OPEN"

                    msg = (
                        f"🤖 AIEM ALERT — ${ticker}\n"
                        f"Confidence: {blended:.0f}/100\n"
                        f"Time: {now_et.strftime('%H:%M ET')} ({time_label})\n"
                        f"Setup: {live_predicted}\n"
                        f"Why: {live_reason[:150]}\n"
                        f"Rank: #{rank} today's picks"
                    )

                    _aiem_send_sms(msg)
                    log.info(f"✅ ALERT FIRED: {ticker} conf={blended:.0f}")

                    # Log alert — prevent double fire
                    cur.execute("""
                        INSERT INTO signal_fire_log
                            (signal_name, ticker, fired_at, metadata)
                        VALUES ('AIEM_ALERT', %s, NOW(), %s)
                        ON CONFLICT DO NOTHING
                    """, (ticker, f"conf={blended:.1f} at {now_et.strftime('%H:%M')}"))

                    conn.commit()

                time.sleep(0.2)

            except Exception as e:
                log.error(f"Open watch error {ticker}: {e}")
                continue

    except Exception as e:
        log.error(f"open_watcher error: {e}")
        if conn:
            conn.rollback()
    finally:
        _put_conn(conn)

# FUNCTION 3: MISSED RUNNER ANALYSIS
# Runs at 4:45 PM ET — the most important learning function
# ─────────────────────────────────────────────────────────────
def aiem_missed_runner_analysis():
    """
    AIEM reviews every stock that moved 20%+ today that it DIDN'T flag.
    Reconstructs what those stocks looked like premarket.
    Writes pattern discoveries to aiem_signal_discoveries.
    This is how AIEM learns what it's missing — the key to getting elite.
    """
    today = date.today()
    log.info(f"=== MISSED RUNNER ANALYSIS {today} ===")

    conn = None
    try:
        conn = _get_conn()
        cur  = conn.cursor()

        # What did AIEM flag today?
        cur.execute("""
            SELECT ticker FROM aiem_predictions WHERE prediction_date = %s
        """, (today,))
        flagged = set(row[0] for row in cur.fetchall())

        # Get ALL stocks that moved 20%+ today
        daily_data = _aiem_get_grouped_daily(today)
        big_movers = []
        for stock in daily_data:
            ticker = stock.get('T', '')
            o      = stock.get('o') or 0
            c      = stock.get('c') or 0
            v      = stock.get('v') or 0
            if o > 0 and c > 0:
                move_pct = ((c - o) / o * 100)
                if move_pct >= 20 and ticker not in flagged:
                    big_movers.append({
                        'ticker':   ticker,
                        'open':     o,
                        'close':    c,
                        'volume':   v,
                        'move_pct': move_pct
                    })

        big_movers.sort(key=lambda x: x['move_pct'], reverse=True)
        log.info(f"Found {len(big_movers)} missed runners (moved 20%+ that AIEM didn't flag)")

        discoveries = 0
        for runner in big_movers[:20]:  # Analyze top 20 missed
            ticker   = runner['ticker']
            move_pct = runner['move_pct']

            try:
                # Get details to understand why it moved
                details = _aiem_get_ticker_details(ticker)
                news    = _aiem_get_news(ticker)
                ohlcv   = _aiem_get_ohlcv(ticker, days=2)

                float_sh = details.get('share_class_shares_outstanding') or 0
                mkt_cap  = details.get('market_cap') or 0

                # Build a description of what AIEM missed
                missed_pattern = []
                if float_sh and float_sh < 10_000_000:
                    missed_pattern.append(f"low_float_{float_sh/1e6:.1f}M")
                if mkt_cap and mkt_cap < 100_000_000:
                    missed_pattern.append(f"nano_cap_{mkt_cap/1e6:.0f}M")
                if news:
                    missed_pattern.append("had_catalyst")
                if runner['volume'] > 1_000_000:
                    missed_pattern.append(f"high_vol_{runner['volume']/1e6:.1f}M_shares")
                if ohlcv and len(ohlcv) >= 2:
                    prev_vol = ohlcv[-2].get('v', 1) if len(ohlcv) >= 2 else 1
                    vol_ratio = runner['volume'] / prev_vol if prev_vol > 0 else 0
                    if vol_ratio >= 3:
                        missed_pattern.append(f"vol_surge_{vol_ratio:.1f}x")

                pattern_str = " | ".join(missed_pattern) if missed_pattern else "unknown_pattern"

                discovery_text = (
                    f"MISSED RUNNER: {ticker} moved +{move_pct:.1f}% today. "
                    f"Pattern: {pattern_str}. "
                    f"Open=${runner['open']:.2f} Close=${runner['close']:.2f}. "
                    f"AIEM did not flag this — learn from it."
                )

                # Write to aiem_signal_discoveries
                cur.execute("""
                    INSERT INTO aiem_signal_discoveries
                        (created_at, ticker, discovery_type, description,
                         confidence_score, validated)
                    VALUES (NOW(), %s, 'MISSED_RUNNER', %s, %s, false)
                """, (ticker, discovery_text, round(move_pct, 1)))

                discoveries += 1
                log.info(f"  MISSED: {ticker} +{move_pct:.1f}% — {pattern_str}")

                time.sleep(0.1)

            except Exception as e:
                log.error(f"Missed runner analysis error {ticker}: {e}")
                continue

        conn.commit()
        log.info(f"✅ Logged {discoveries} missed runner discoveries")

        # Send summary SMS to Joel
        if big_movers:
            top3 = big_movers[:3]
            msg  = (
                f"🤖 AIEM EOD REVIEW — {today}\n"
                f"Missed runners: {len(big_movers)}\n"
                f"Top missed:\n"
            )
            for r in top3:
                msg += f"  ${r['ticker']} +{r['move_pct']:.1f}%\n"
            msg += f"AIEM analyzing patterns to catch these tomorrow."
            _aiem_send_sms(msg)

    except Exception as e:
        log.error(f"missed_runner_analysis error: {e}")
        if conn:
            conn.rollback()
    finally:
        _put_conn(conn)

# FUNCTION 4: GRADE OUTCOMES
# Runs at 4:30 PM ET
# ─────────────────────────────────────────────────────────────
def aiem_grade_outcomes():
    """Grade today's predictions with real EOD prices."""
    today = date.today()
    log.info(f"=== GRADING OUTCOMES {today} ===")

    conn = None
    try:
        conn = _get_conn()
        cur  = conn.cursor()

        cur.execute("""
            SELECT p.ticker, p.confidence_score
            FROM aiem_predictions p
            LEFT JOIN aiem_prediction_outcomes o
                ON p.ticker = o.ticker AND p.prediction_date = o.prediction_date
            WHERE p.prediction_date = %s AND o.id IS NULL
        """, (today,))
        ungraded = cur.fetchall()

        if not ungraded:
            log.info("Nothing to grade today")
            return

        graded = 0
        wins   = 0
        for ticker, conf in ungraded:
            try:
                snap        = _aiem_get_snapshot(ticker)
                day         = snap.get('day', {})
                entry_price = day.get('o') or 0
                t1_price    = day.get('c') or 0

                if not entry_price or not t1_price:
                    continue

                t1_return = ((t1_price - entry_price) / entry_price * 100)
                win_t1    = t1_return > 0

                cur.execute("""
                    INSERT INTO aiem_prediction_outcomes
                        (prediction_date, ticker, entry_price, t1_price, t1_return, graded_at)
                    VALUES (%s, %s, %s, %s, %s, NOW())
                    ON CONFLICT (prediction_date, ticker) DO UPDATE
                        SET t1_price=EXCLUDED.t1_price,
                            t1_return=EXCLUDED.t1_return,
                            graded_at=NOW()
                """, (today, ticker, entry_price, t1_price, round(t1_return, 4)))

                graded += 1
                if win_t1:
                    wins += 1
                label = "WIN ✅" if win_t1 else "LOSS ❌"
                log.info(f"  {ticker}: {t1_return:+.1f}% {label}")
                time.sleep(0.1)

            except Exception as e:
                log.error(f"Grade error {ticker}: {e}")

        conn.commit()
        win_rate = (wins / graded * 100) if graded > 0 else 0
        log.info(f"✅ Graded {graded} predictions — win rate today: {win_rate:.1f}%")

        # Update T3/T5 for older predictions
        _aiem_grade_t3_t5(conn)

    except Exception as e:
        log.error(f"grade_outcomes error: {e}")
        if conn:
            conn.rollback()
    finally:
        _put_conn(conn)


def _aiem_grade_t3_t5(conn):
    """Update T3 and T5 returns for predictions 3 and 5 days old."""
    today = date.today()
    try:
        cur = conn.cursor()
        for days, col_price, col_return, col_win in [
            (3, 't3_price', 't3_return', 'win_t3'),
            (5, 't5_price', 't5_return', 'win_t5'),
        ]:
            target_date = today - timedelta(days=days)
            cur.execute(f"""
                SELECT ticker, entry_price FROM aiem_prediction_outcomes
                WHERE prediction_date = %s AND {col_price} IS NULL AND entry_price IS NOT NULL
            """, (target_date,))
            rows = cur.fetchall()
            for ticker, entry_price in rows:
                try:
                    snap  = _aiem_get_snapshot(ticker)
                    price = snap.get('day', {}).get('c') or 0
                    if price and entry_price:
                        ret = (price - float(entry_price)) / float(entry_price) * 100
                        cur.execute(f"""
                            UPDATE aiem_prediction_outcomes
                            SET {col_price}=%s, {col_return}=%s, {col_win}=%s
                            WHERE prediction_date=%s AND ticker=%s
                        """, (price, round(ret, 4), ret > 0, target_date, ticker))
                        log.info(f"  T{days} graded {ticker}: {ret:+.1f}%")
                    time.sleep(0.1)
                except Exception as e:
                    log.error(f"T{days} grade error {ticker}: {e}")
        conn.commit()
    except Exception as e:
        log.error(f"grade_t3_t5 error: {e}")

# FUNCTION 5: NIGHTLY LEARN
# Runs at 6:00 PM ET — AIEM updates its own brain
# ─────────────────────────────────────────────────────────────
def aiem_nightly_learn():
    """
    AIEM's learning engine. Reviews outcomes vs predictions.
    Updates signal_trust_weights — winning signals get boosted,
    losing signals get reduced. Also mines missed runner patterns.
    Over weeks AIEM discovers its own proprietary microcap signals.
    """
    today = date.today()
    log.info(f"=== NIGHTLY LEARN {today} ===")

    conn = None
    try:
        conn = _get_conn()
        cur  = conn.cursor()

        # ── Join predictions to outcomes — last 60 days
        cur.execute("""
            SELECT
                p.signal_basis,
                p.confidence_score,
                p.ticker,
                o.t1_return,
                o.win_t3,
                o.t3_return,
                o.win_t5,
                o.t5_return
            FROM aiem_predictions p
            JOIN aiem_prediction_outcomes o
                ON p.ticker = o.ticker AND p.prediction_date = o.prediction_date
            WHERE p.prediction_date >= %s
              AND o.t1_return IS NOT NULL
        """, (today - timedelta(days=60),))
        rows = cur.fetchall()

        if not rows:
            log.info("No graded outcomes yet — come back tomorrow")
            return

        # Tally per signal
        tallies = {}
        for sig_basis, conf, ticker, t1_ret, win_t3, t3_ret, win_t5, t5_ret in rows:
            if not sig_basis:
                continue
            # Primary learning target: T3 win if available, else T1
            win = bool(win_t3) if win_t3 is not None else (float(t1_ret or 0) > 0)
            ret = float(t3_ret or t1_ret or 0)

            for sig in [s.strip() for s in sig_basis.split(',') if s.strip()]:
                if sig not in tallies:
                    tallies[sig] = {'wins': 0, 'total': 0, 'total_ret': 0.0}
                tallies[sig]['total']     += 1
                tallies[sig]['total_ret'] += ret
                if win:
                    tallies[sig]['wins'] += 1

        # Update signal_trust_weights
        updated = 0
        for sig_name, t in tallies.items():
            n        = t['total']
            win_rate = t['wins'] / n if n > 0 else 0.5
            avg_ret  = t['total_ret'] / n if n > 0 else 0

            # Trust formula: conservative early (low n), confident later (high n)
            sample_conf  = min(1.0, n / 30)
            raw_trust    = 0.5 + (win_rate - 0.5) * 2
            trust_weight = 1.0 + (raw_trust - 0.5) * sample_conf
            trust_weight = max(0.3, min(2.0, trust_weight))  # Clamp 0.3–2.0

            cur.execute("""
                INSERT INTO signal_trust_weights
                    (signal_name, context_bucket, rolling_win_rate,
                     n_outcomes_observed, trust_weight, last_updated_at)
                VALUES (%s, 'AIEM_MICROCAP', %s, %s, %s, NOW())
                ON CONFLICT (signal_name, context_bucket) DO UPDATE
                    SET rolling_win_rate    = EXCLUDED.rolling_win_rate,
                        n_outcomes_observed = EXCLUDED.n_outcomes_observed,
                        trust_weight        = EXCLUDED.trust_weight,
                        last_updated_at     = NOW()
            """, (sig_name, round(win_rate, 4), n, round(trust_weight, 4)))

            updated += 1
            log.info(f"  {sig_name}: win={win_rate:.1%} n={n} "
                     f"trust={trust_weight:.3f} avg_ret={avg_ret:+.1f}%")

        # ── Mine missed runner patterns into signal discoveries
        cur.execute("""
            SELECT description FROM aiem_signal_discoveries
            WHERE discovery_type = 'MISSED_RUNNER'
              AND created_at >= NOW() - INTERVAL '7 days'
              AND validated = false
            LIMIT 50
        """)
        missed_rows = cur.fetchall()

        # Count pattern frequencies across missed runners
        pattern_freq = {}
        for (desc,) in missed_rows:
            for pattern in ['low_float', 'nano_cap', 'had_catalyst',
                           'high_vol', 'vol_surge', 'squeeze']:
                if pattern in desc:
                    pattern_freq[pattern] = pattern_freq.get(pattern, 0) + 1

        # If a pattern appears in 3+ missed runners — AIEM notes it
        new_insights = []
        for pattern, freq in pattern_freq.items():
            if freq >= 3:
                insight = (f"PATTERN DISCOVERY: '{pattern}' appeared in "
                          f"{freq} missed runners this week — "
                          f"AIEM should weight this signal higher")
                new_insights.append(insight)
                log.info(f"  🔍 {insight}")

                # Write as research insight
                cur.execute("""
                    INSERT INTO aiem_research_insights
                        (created_at, insight_type, summary, confidence_score)
                    VALUES (NOW(), 'MISSED_PATTERN', %s, %s)
                """, (insight, min(95, freq * 15)))

        conn.commit()
        log.info(f"✅ nightly_learn: updated {updated} signal weights, "
                 f"{len(new_insights)} pattern discoveries")

        # Send nightly summary SMS
        top_sigs = sorted(tallies.items(),
                         key=lambda x: x[1]['wins']/max(x[1]['total'],1),
                         reverse=True)[:3]
        sig_summary = ", ".join(
            f"{s}({d['wins']}/{d['total']})" for s, d in top_sigs
        )
        msg = (
            f"🤖 AIEM NIGHTLY REPORT — {today}\n"
            f"Signals updated: {updated}\n"
            f"Top signals: {sig_summary}\n"
            f"Pattern discoveries: {len(new_insights)}\n"
            f"Ready for tomorrow's scan."
        )
        _aiem_send_sms(msg)

    except Exception as e:
        log.error(f"nightly_learn error: {e}")
        if conn:
            conn.rollback()
    finally:
        _put_conn(conn)

# MISSED MORNING DETECTOR
# Runs at 9:45 AM — safety net if premarket scan failed
# ─────────────────────────────────────────────────────────────
def aiem_missed_morning_check():
    """
    If AIEM has no predictions for today by 9:45am —
    something went wrong. Run emergency scan immediately.
    """
    conn = None
    try:
        conn = _get_conn()
        cur  = conn.cursor()
        cur.execute("""
            SELECT COUNT(*) FROM aiem_predictions
            WHERE prediction_date = %s
        """, (date.today(),))
        count = cur.fetchone()[0]
        if count == 0:
            log.warning("⚠️ NO PREDICTIONS FOR TODAY — running emergency scan")
            _aiem_send_sms("⚠️ AIEM: No premarket predictions found at 9:45am — running emergency scan now")
            aiem_premarket_scan()
        else:
            log.info(f"Morning check OK — {count} predictions on file")
    except Exception as e:
        log.error(f"missed_morning_check error: {e}")
    finally:
        _put_conn(conn)

    log.info("Standalone process — independent of Flask")
    log.info("=" * 60)

    # Verify connections on startup
    try:
        _init_pool()

        log.error(f"❌ DB pool failed: {e}")
        return

    if not POLYGON_API_KEY:
        log.error("❌ POLYGON_API_KEY not set")
        return

    scheduler = BlockingScheduler(
        timezone=ET,
        job_defaults={
            'coalesce':          True,
            'max_instances':     1,
            'misfire_grace_time': 3600  # 1 hour grace — never miss a morning
        }
    )

    # ── Premarket scan: every 15 min 7:00–9:30 AM
    scheduler.add_job(
        aiem_premarket_scan,

    )

    scheduler.add_job(
        aiem_open_watcher,
        'cron', hour='9,10', minute='*/5',
        id='aiem_open_watch', replace_existing=True
    )

    # ── Missed morning safety net: 9:45 AM
    scheduler.add_job(
        aiem_missed_morning_check,
        'cron', hour=9, minute=45,
        id='aiem_morning_check', replace_existing=True
    )

    # ── Grade outcomes: 4:30 PM
    scheduler.add_job(
        aiem_grade_outcomes,
        'cron', hour=16, minute=30,
        id='aiem_grade', replace_existing=True
    )

    # ── Missed runner analysis: 4:45 PM
    scheduler.add_job(
        aiem_missed_runner_analysis,
        'cron', hour=16, minute=45,
        id='aiem_missed', replace_existing=True
    )

    # ── Nightly learn: 6:00 PM
    scheduler.add_job(
        aiem_nightly_learn,
        'cron', hour=18, minute=0,
        id='aiem_learn', replace_existing=True
    )

    log.info("Scheduler jobs registered:")
    log.info("  7:00–9:30 AM  → premarket_scan (every 15min)")
    log.info("  9:30–10:30 AM → open_watcher (every 5min)")
    log.info("  9:45 AM       → missed_morning_check")
    log.info("  4:30 PM       → grade_outcomes")
    log.info("  4:45 PM       → missed_runner_analysis")
    log.info("  6:00 PM       → nightly_learn")
    log.info("AIEM is live. Watching markets autonomously.")

    try:
        scheduler.start()
    except (KeyboardInterrupt, SystemExit):
        log.info("AIEM autonomous engine stopped")


if __name__ == '__main__':

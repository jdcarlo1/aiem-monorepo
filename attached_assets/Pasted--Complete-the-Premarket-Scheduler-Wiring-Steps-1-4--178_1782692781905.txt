```python
def _run_premarket_open_tracker():
    """
    Runs every 5 min, 9:30-10:00 AM ET. Records a live snapshot for
    each premarket gap candidate and evaluates the combined decision.
    """
    import os
    import premarket_scanner as ps
    import opening_snapshot_tracker as ost
    import premarket_open_trader as pot

    db_url = os.environ.get("AIEM_DATABASE_URL") or os.environ.get("DATABASE_URL")
    if not db_url:
        print("[premarket_open_tracker] DATABASE_URL not set, skipping run")
        return

    # STEP 4 connects here: pull real candidates instead of an empty list
    universe = list(_MICRO_CAP_UNIVERSE)  # reuse existing universe, or your full ticker list
    candidates = ps.scan_premarket_universe(db_url, universe, min_gap_pct=3.0, min_premarket_volume=50000)

    for c in candidates:
        ticker = c["ticker"]
        try:
            # Same live-quote pattern as gamma scanner — CONFIRM this
            # matches Step 1's real output before trusting it:
            tk = _TdTicker(ticker)
            fi = tk.fast_info
            price = _sf(getattr(fi, "last_price", None))
            volume = _si(getattr(fi, "last_volume", None) or 0)

            if not price:
                continue

            ost.record_snapshot(db_url, ticker, price, volume)
            result = pot.evaluate_ticker(db_url, ticker, premarket_gap_pct=c["gap_pct"])
            print(f"[premarket_open_tracker] {ticker}: {result['decision']}")
        except Exception as e:
            print(f"[premarket_open_tracker] error on {ticker}: {e}")
            continue
```
